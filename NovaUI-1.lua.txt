local function create(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props or {}) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function corner(inst, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius or 8)
	c.Parent = inst
	return c
end

local function stroke(inst, color, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color or Color3.new(1, 1, 1)
	s.Thickness = thickness or 1
	s.Parent = inst
	return s
end

local TWEEN_FAST   = TweenInfo.new(0.10, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local TWEEN_NORMAL = TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local TWEEN_SLOW   = TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function tween(inst, props, time)
	if VeloxUI.ReducedMotion then
		pcall(function()
			for k, v in pairs(props) do
				inst[k] = v
			end
		end)
		return nil
	end
	local info = TWEEN_NORMAL
	if time then
		if time <= 0.12 then
			info = TWEEN_FAST
		elseif time >= 0.25 then
			info = TWEEN_SLOW
		else
			info = TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
		end
	end
	local ok, t = pcall(function()
		local tw = TweenService:Create(inst, info, props)
		tw:Play()
		return tw
	end)
	if ok then
		return t
	end
end

local function shade(color, delta)
	local h, s, v = color:ToHSV()
	v = math.clamp(v + delta, 0, 1)
	return Color3.fromHSV(h, s, v)
end

local function safeCall(fn, ...)
	if not fn then return end
	local ok, err = pcall(fn, ...)
	if not ok then
		warn("[VeloxUI] Callback error: " .. tostring(err))
	end
	return ok
end

local function wireButtonFeel(btn, getBase)
	local function reset()
		tween(btn, { BackgroundColor3 = getBase() }, 0.12)
	end
	btn.MouseEnter:Connect(function()
		tween(btn, { BackgroundColor3 = shade(getBase(), 0.08) }, 0.12)
	end)
	btn.MouseLeave:Connect(reset)
	btn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			tween(btn, { BackgroundColor3 = shade(getBase(), -0.08) }, 0.08)
		end
	end)
	btn.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			reset()
		end
	end)
end

local function isImageIcon(icon)
	if type(icon) ~= "string" then return false end
	return icon:match("^rbxassetid://") ~= nil or icon:match("^%d+$") ~= nil
end

local function renderIcon(parent, icon, size, color)
	if icon and isImageIcon(icon) then
		local assetId = icon:match("^%d+$") and ("rbxassetid://" .. icon) or icon
		local ok, inst = pcall(create, "ImageLabel", {
			BackgroundTransparency = 1,
			Size = UDim2.new(1, 0, 1, 0),
			Image = assetId,
			ImageColor3 = color,
			ScaleType = Enum.ScaleType.Fit,
		}, parent)
		if ok then
			return inst
		end
	end
	return create("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 1, 0),
		Font = Enum.Font.GothamBold,
		Text = icon or "",
		TextColor3 = color,
		TextSize = size or 16,
	}, parent)
end

local function tintIcon(inst, color, time)
	if not inst then return end
	if inst:IsA("ImageLabel") then
		tween(inst, { ImageColor3 = color }, time)
	else
		tween(inst, { TextColor3 = color }, time)
	end
end

local Viewport = {}
do
	local listeners = {}

	local function getCamera()
		return Workspace.CurrentCamera
	end

	function Viewport.Get()
		local cam = getCamera()
		if cam then
			return cam.ViewportSize
		end
		return Vector2.new(1280, 720)
	end

	function Viewport.IsCompact()
		local size = Viewport.Get()
		local touchOnly = UserInputService.TouchEnabled and not UserInputService.MouseEnabled
		return touchOnly or size.X < 650
	end

	local function fire()
		for _, fn in ipairs(listeners) do
			pcall(fn)
		end
	end

	function Viewport.OnChange(fn)
		table.insert(listeners, fn)
		return function()
			local idx = table.find(listeners, fn)
			if idx then
				table.remove(listeners, idx)
			end
		end
	end

	local function hook(cam)
		if cam then
			cam:GetPropertyChangedSignal("ViewportSize"):Connect(fire)
		end
	end

	hook(getCamera())
	Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
		hook(getCamera())
		fire()
	end)
end

local InputBus = {}
do
	local captureCallback
	local toggleBindings = {}

	UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end

		if captureCallback then
			local cb = captureCallback
			captureCallback = nil
			cb(input.KeyCode)
			return
		end

		if gameProcessed then
			return
		end

		for _, binding in ipairs(toggleBindings) do
			if binding.keycode == input.KeyCode then
				safeCall(binding.fn)
			end
		end
	end)

	function InputBus.CaptureNextKey(callback)
		captureCallback = callback
	end

	function InputBus.CancelCapture(callback)
		if captureCallback == callback then
			captureCallback = nil
		end
	end

	function InputBus.RegisterToggleKey(keycode, fn)
		local entry = { keycode = keycode, fn = fn }
		table.insert(toggleBindings, entry)
		return function()
			local idx = table.find(toggleBindings, entry)
			if idx then
				table.remove(toggleBindings, idx)
			end
		end
	end
end

local DragBus = {}
do
	local activeMove, activeEnd

	UserInputService.InputChanged:Connect(function(input)
		if activeMove and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			activeMove(input)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if activeEnd and (input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch) then
			local fn = activeEnd
			activeMove, activeEnd = nil, nil
			fn(input)
		end
	end)

	function DragBus.Begin(onMove, onEnd)
		activeMove = onMove
		activeEnd = onEnd or function() end
	end

	function DragBus.Cancel()
		activeMove, activeEnd = nil, nil
	end
end

local function makeDraggable(handle, target)
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			local startPos = target.Position
			local startInput = input.Position
			DragBus.Begin(function(moveInput)
				local delta = moveInput.Position - startInput
				target.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + delta.X,
					startPos.Y.Scale, startPos.Y.Offset + delta.Y
				)
			end)
		end
	end)
end

local NotifHolder

local function ensureNotifHolder()
	if NotifHolder and NotifHolder.Parent then
		return NotifHolder
	end

	local gui = create("ScreenGui", {
		Name = "VeloxUI_Notifications",
		ResetOnSpawn = false,
		IgnoreGuiInset = true,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	})

	local ok = pcall(function()
		gui.Parent = game:GetService("CoreGui")
	end)
	if not ok then
		gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
	end

	local function holderSize()
		local vp = Viewport.Get()
		return UDim2.new(0, math.min(300, vp.X - 32), 1, -32)
	end

	NotifHolder = create("Frame", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, -16, 1, -16),
		Size = holderSize(),
	}, gui)

	create("UIListLayout", {
		Padding = UDim.new(0, 8),
		VerticalAlignment = Enum.VerticalAlignment.Bottom,
		HorizontalAlignment = Enum.HorizontalAlignment.Right,
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, NotifHolder)

	Viewport.OnChange(function()
		if NotifHolder and NotifHolder.Parent then
			NotifHolder.Size = holderSize()
		end
	end)

	return NotifHolder
end

function VeloxUI:Notify(config)
	config = config or {}
	local theme = self.CurrentTheme
	local holder = ensureNotifHolder()

	local frame = create("Frame", {
		BackgroundColor3 = theme.Element,
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		ClipsDescendants = true,
	}, holder)
	corner(frame, 10)
	stroke(frame, theme.Stroke, 1)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10),
		PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 12),
	}, frame)

	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -12),
		Position = UDim2.new(0, 0, 0, 6),
		BackgroundColor3 = theme.Accent,
		BackgroundTransparency = 1,
	}, frame)
	corner(accentBar, 2)

	local title = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 8, 0, 0),
		Size = UDim2.new(1, -8, 0, 18),
		Font = Enum.Font.GothamBold,
		Text = config.Title or "Notification",
		TextColor3 = theme.Text,
		TextTransparency = 1,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, frame)

	local content = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 8, 0, 20),
		Size = UDim2.new(1, -8, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		Font = Enum.Font.Gotham,
		Text = config.Content or "",
		TextColor3 = theme.SubText,
		TextTransparency = 1,
		TextSize = 13,
		TextWrapped = true,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, frame)

	tween(frame, { BackgroundTransparency = 0 }, 0.18)
	tween(accentBar, { BackgroundTransparency = 0 }, 0.18)
	tween(title, { TextTransparency = 0 }, 0.18)
	tween(content, { TextTransparency = 0 }, 0.18)

	task.delay(config.Duration or 3, function()
		if not (frame and frame.Parent) then return end
		tween(frame, { BackgroundTransparency = 1 }, 0.2)
		tween(accentBar, { BackgroundTransparency = 1 }, 0.2)
		tween(title, { TextTransparency = 1 }, 0.2)
		local hideTween = tween(content, { TextTransparency = 1 }, 0.2)
		if hideTween then
			hideTween.Completed:Connect(function()
				if frame and frame.Parent then
					frame:Destroy()
				end
			end)
		elseif frame and frame.Parent then
			frame:Destroy()
		end
	end)
end

		}
	end
end

local function defaultWindowSize(compact)
	local vp = Viewport.Get()
	if compact then
		local w = math.clamp(vp.X - 24, 220, 460)
		local h = math.clamp(vp.Y - 80, 200, 640)
		return UDim2.fromOffset(math.floor(w), math.floor(h))
	else
		local w = math.clamp(vp.X * 0.55, 560, 760)
		local h = math.clamp(vp.Y * 0.68, 420, 640)
		return UDim2.fromOffset(math.floor(w), math.floor(h))
	end
end

			Font = Enum.Font.GothamBold,
			Text = sf.Title or "",
			TextColor3 = theme.Text,
			TextSize = metrics.FontDesc + 1,
			TextXAlignment = Enum.TextXAlignment.Left,
		}, card)
		create("TextLabel", {
			BackgroundTransparency = 1,
			Position = UDim2.new(0, 0, 0, 18),
			Size = UDim2.new(1, 0, 0, 14),
			Font = Enum.Font.Gotham,
			Text = sf.SubTitle or "",
			TextColor3 = theme.SubText,
			TextSize = metrics.FontDesc - 1,
			TextXAlignment = Enum.TextXAlignment.Left,
		}, card)
		local bar = create("Frame", {
			Position = UDim2.new(0, 0, 1, -12),
			Size = UDim2.new(0.4, 0, 0, 3),
			BackgroundColor3 = theme.Accent,
		}, card)
		corner(bar, 2)
	end

	local content = create("Frame", {
		BackgroundTransparency = 1,
	}, main)

	if footerHeight > 0 then
		local footer = create("Frame", {
			Size = UDim2.new(1, 0, 0, footerHeight),
			Position = UDim2.new(0, 0, 1, -footerHeight),
			BackgroundColor3 = theme.Sidebar,
			BorderSizePixel = 0,
		}, main)
		local leftText = table.concat(config.Footer.Left or {}, "   •   ")
		create("TextLabel", {
			BackgroundTransparency = 1,
			Position = UDim2.new(0, 14, 0, 0),
			Size = UDim2.new(0.65, 0, 1, 0),
			Font = Enum.Font.Gotham,
			Text = leftText,
			TextColor3 = theme.SubText,
			TextSize = metrics.FontDesc - 1,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextTruncate = Enum.TextTruncate.AtEnd,
		}, footer)
		create("TextLabel", {
			BackgroundTransparency = 1,
			AnchorPoint = Vector2.new(1, 0.5),
			Position = UDim2.new(1, -14, 0.5, 0),
			Size = UDim2.new(0.35, 0, 1, 0),
			Font = Enum.Font.GothamBold,
			Text = config.Footer.Right or "",
			TextColor3 = theme.Accent,
			TextSize = metrics.FontDesc - 1,
			TextXAlignment = Enum.TextXAlignment.Right,
		}, footer)
	end

	local overlay = create("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		ZIndex = 50,
	}, gui)

	local popupCatcher = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		Visible = false,
		ZIndex = 52,
	}, overlay)

	local activePopup = nil

	local function closeActivePopup()
		if activePopup then
			activePopup.Visible = false
			activePopup = nil
		end
		popupCatcher.Visible = false
	end

	local function openPopup(popupFrame, anchor, width, height)
		if activePopup and activePopup ~= popupFrame then
			activePopup.Visible = false
		end
		local vp = Viewport.Get()
		local anchorPos = anchor.AbsolutePosition
		local anchorSize = anchor.AbsoluteSize
		local x = anchorPos.X
		local y = anchorPos.Y + anchorSize.Y + 4
		if x + width > vp.X then
			x = math.max(vp.X - width - 8, 0)
		end
		if y + height > vp.Y then
			y = anchorPos.Y - height - 4
			if y < 0 then
				y = math.max(vp.Y - height - 8, 0)
			end
		end
		popupFrame.Position = UDim2.fromOffset(x, y)
		popupFrame.Visible = true
		popupCatcher.Visible = true
		activePopup = popupFrame
	end

	popupCatcher.MouseButton1Click:Connect(closeActivePopup)

	local drawerOpen = false

	local function applyLayoutMode()
		local bottomInset = topBarHeight + footerHeight
		if compact then
			local width = math.min(sidebarWidth, Viewport.Get().X - 40)
			sidebar.Position = UDim2.new(0, drawerOpen and 0 or -sidebarWidth, 0, topBarHeight)
			sidebar.Size = UDim2.new(0, width, 1, -bottomInset)
			sidebar.ZIndex = 20
			content.Position = UDim2.new(0, 0, 0, topBarHeight)
			content.Size = UDim2.new(1, 0, 1, -bottomInset)
			backdrop.Position = UDim2.new(0, 0, 0, topBarHeight)
			backdrop.Size = UDim2.new(1, 0, 1, -bottomInset)
			hamburger.Visible = true
		else
			sidebar.Position = UDim2.new(0, 0, 0, topBarHeight)
			sidebar.Size = UDim2.new(0, sidebarWidth, 1, -bottomInset)
			sidebar.ZIndex = 1
			content.Position = UDim2.new(0, sidebarWidth, 0, topBarHeight)
			content.Size = UDim2.new(1, -sidebarWidth, 1, -bottomInset)
			backdrop.Position = UDim2.new(0, 0, 0, topBarHeight)
			backdrop.Size = UDim2.new(1, 0, 1, -bottomInset)
			hamburger.Visible = false
			drawerOpen = false
			backdrop.Visible = false
		end
	end

	local function setDrawerOpen(open)
		if not compact then return end
		drawerOpen = open
		tween(sidebar, {
			Position = UDim2.new(0, open and 0 or -sidebarWidth, 0, topBarHeight),
		}, 0.2)
		if open then
			backdrop.Visible = true
			backdrop.BackgroundTransparency = 1
			tween(backdrop, { BackgroundTransparency = 0.45 }, 0.2)
		else
			local hideTween = tween(backdrop, { BackgroundTransparency = 1 }, 0.2)
			if hideTween then
				hideTween.Completed:Connect(function()
					if not drawerOpen then
						backdrop.Visible = false
					end
				end)
			else
				backdrop.Visible = false
			end
		end
	end

	hamburger.MouseButton1Click:Connect(function()
		setDrawerOpen(not drawerOpen)
	end)
	backdrop.MouseButton1Click:Connect(function()
		setDrawerOpen(false)
	end)

	applyLayoutMode()

	closeBtn.MouseButton1Click:Connect(function()
		gui:Destroy()
	end)

	local minimized = false
	local fullSize = main.Size
	minBtn.MouseButton1Click:Connect(function()
		minimized = not minimized
		if minimized then
			tween(main, { Size = UDim2.new(fullSize.X.Scale, fullSize.X.Offset, 0, topBarHeight) }, 0.22)
		else
			tween(main, { Size = fullSize }, 0.22)
		end
	end)

	local isMaximized = false
	local preMaxSize, preMaxPos
	maxBtn.MouseButton1Click:Connect(function()
		if minimized then return end
		isMaximized = not isMaximized
		if isMaximized then
			preMaxSize, preMaxPos = main.Size, main.Position
			local vp = Viewport.Get()
			local margin = compact and 6 or 20
			fullSize = UDim2.fromOffset(vp.X - margin * 2, vp.Y - margin * 2)
			tween(main, { Size = fullSize, Position = UDim2.fromOffset(margin, margin) }, 0.22)
		else
			fullSize = preMaxSize or fullSize
			tween(main, { Size = fullSize, Position = preMaxPos or main.Position }, 0.22)
		end
	end)

	local function clampMainPosition()
		local vp = Viewport.Get()
		local size = main.Size
		local pos = main.Position
		local maxX = math.max(vp.X - size.X.Offset, 0)
		local maxY = math.max(vp.Y - size.Y.Offset, 0)
		local x = math.clamp(pos.X.Offset, 0, maxX)
		local y = math.clamp(pos.Y.Offset, 0, maxY)
		if x ~= pos.X.Offset or y ~= pos.Y.Offset then
			main.Position = UDim2.new(pos.X.Scale, x, pos.Y.Scale, y)
		end
	end

	local unregisterToggleKey
	if config.ToggleKey then
		unregisterToggleKey = InputBus.RegisterToggleKey(config.ToggleKey, function()
			main.Visible = not main.Visible
		end)
	end

	local unsubscribeViewport = Viewport.OnChange(function()
		local wasCompact = compact
		compact = Viewport.IsCompact()
		if compact ~= wasCompact then
			applyLayoutMode()
		end
		if not usesCustomSize and not minimized and not isMaximized then
			fullSize = defaultWindowSize(compact)
			main.Size = fullSize
		end
		clampMainPosition()
	end)

	local window = setmetatable({
		_theme = theme,
		_metrics = metrics,
		_gui = gui,
		_main = main,
		_tabList = tabList,
		_content = content,
		_tabs = {},
		_overlay = overlay,
		_openPopup = openPopup,
		_closeActivePopup = closeActivePopup,
		_setDrawerOpen = setDrawerOpen,
		_unsubscribeViewport = unsubscribeViewport,
		_unregisterToggleKey = unregisterToggleKey,
		_titleLabel = titleLabel,
		_subTitleLabel = subTitleLabel,
		_destroyed = false,
	}, WindowMeta)

	return window
end

function WindowMeta:Toggle()
	if self._destroyed then return end
	self._main.Visible = not self._main.Visible
end

function WindowMeta:SetTitle(text)
	if self._titleLabel then
		self._titleLabel.Text = text or ""
	end
end

function WindowMeta:SetSubTitle(text)
	if self._subTitleLabel then
		self._subTitleLabel.Text = text or ""
	end
end

function WindowMeta:Destroy()
	if self._destroyed then return end
	self._destroyed = true
	if self._unsubscribeViewport then
		self._unsubscribeViewport()
	end
	if self._unregisterToggleKey then
		self._unregisterToggleKey()
	end
	self._closeActivePopup()
	self._gui:Destroy()
end

function WindowMeta:CreateTab(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics

	local tabButton = create("TextButton", {

		BackgroundColor3 = theme.Accent,
		BackgroundTransparency = 1,
		Text = "",

	corner(tabButton, metrics.Radius - 2)

	local iconHolder = create("Frame", {
		Position = UDim2.new(0, 6, 0.5, -13),
		Size = UDim2.fromOffset(26, 26),
		BackgroundColor3 = theme.Element,
	}, tabButton)
	corner(iconHolder, 7)
	local iconLabel = renderIcon(iconHolder, config.Icon, 13, theme.SubText)

	local tabLabel = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 40, 0, 0),
		Size = UDim2.new(1, -46, 1, 0),
		Font = Enum.Font.GothamMedium,
		Text = config.Title or "Tab",
		TextColor3 = theme.SubText,
		TextSize = metrics.FontTitle,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, tabButton)

	local page = create("ScrollingFrame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 3,
		CanvasSize = UDim2.new(0, 0, 0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		Visible = false,
	}, self._content)
	create("UIListLayout", {
		Padding = UDim.new(0, 12),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, page)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 14), PaddingBottom = UDim.new(0, 14),
		PaddingLeft = UDim.new(0, 14), PaddingRight = UDim.new(0, 14),
	}, page)

	local tab = setmetatable({
		_window = self,
		_page = page,
		_theme = theme,
		_metrics = metrics,
		_button = tabButton,
		_label = tabLabel,
		_iconHolder = iconHolder,
		_icon = iconLabel,
		_selected = false,
		_openRow = nil,
	}, TabMeta)

	tabButton.MouseButton1Click:Connect(function()
		tab:Select()
	end)
	tabButton.MouseEnter:Connect(function()
		if not tab._selected then
			tween(tabButton, { BackgroundTransparency = 0.92 }, 0.12)
		end
	end)
	tabButton.MouseLeave:Connect(function()
		if not tab._selected then
			tween(tabButton, { BackgroundTransparency = 1 }, 0.12)
		end
	end)

	table.insert(self._tabs, tab)
	if #self._tabs == 1 then
		tab:Select()
	end

	return tab
end

function TabMeta:Select()
	local window = self._window
	for _, t in ipairs(window._tabs) do
		t._selected = false
		t._page.Visible = false
		tween(t._button, { BackgroundTransparency = 1 }, 0.15)
		tween(t._label, { TextColor3 = t._theme.SubText }, 0.15)
		tween(t._iconHolder, { BackgroundColor3 = t._theme.Element }, 0.15)
		tintIcon(t._icon, t._theme.SubText, 0.15)
	end

	self._selected = true
	self._page.Visible = true
	tween(self._button, { BackgroundTransparency = 0.85 }, 0.15)
	tween(self._label, { TextColor3 = self._theme.Text }, 0.15)
	tween(self._iconHolder, { BackgroundColor3 = self._theme.Accent }, 0.15)
	tintIcon(self._icon, Color3.new(1, 1, 1), 0.15)

	if window._setDrawerOpen then
		window._setDrawerOpen(false)
	end
end

function TabMeta:Destroy()
	local window = self._window
	if not window then return end
	local idx = table.find(window._tabs, self)
	if idx then
		table.remove(window._tabs, idx)
	end
	if self._button then self._button:Destroy() end
	if self._page then self._page:Destroy() end
	if self._selected and #window._tabs > 0 then
		window._tabs[1]:Select()
	end
end

function TabMeta:CreateCard(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local half = config.Half and not metrics.Compact
	local parent = self._page

	local container
	if half then
		if self._openRow and self._openRowSlots and self._openRowSlots < 2 then
			container = self._openRow
			self._openRowSlots = self._openRowSlots + 1
			if self._openRowSlots >= 2 then
				self._openRow = nil
			end
		else
			local row = create("Frame", {
				Size = UDim2.new(1, 0, 0, 0),
				AutomaticSize = Enum.AutomaticSize.Y,
				BackgroundTransparency = 1,
			}, parent)
			create("UIListLayout", {
				FillDirection = Enum.FillDirection.Horizontal,
				Padding = UDim.new(0, 12),
				SortOrder = Enum.SortOrder.LayoutOrder,
			}, row)
			self._openRow = row
			self._openRowSlots = 1
			container = row
		end
	else
		self._openRow = nil
		self._openRowSlots = nil
		container = parent
	end

	local width = half and UDim2.new(0.5, -6, 0, 0) or UDim2.new(1, 0, 0, 0)

	local card = create("Frame", {
		Size = width,
		AutomaticSize = Enum.AutomaticSize.Y,
		BackgroundColor3 = theme.Element,
	}, container)
	corner(card, metrics.Radius)
	stroke(card, theme.Stroke, 1)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 14), PaddingBottom = UDim.new(0, 14),
		PaddingLeft = UDim.new(0, 14), PaddingRight = UDim.new(0, 14),
	}, card)
	create("UIListLayout", {
		Padding = UDim.new(0, 10),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, card)

	if config.Title then
		local header = create("Frame", {
			Size = UDim2.new(1, 0, 0, 22),
			BackgroundTransparency = 1,
		}, card)
		local textStart = 0

				BackgroundTransparency = 1,
				Size = UDim2.new(0, 20, 1, 0),
			}, header)
			renderIcon(iconWrap, config.Icon, metrics.FontTitle + 2, theme.Accent)
			textStart = 24
		end

			BackgroundTransparency = 1,
			Position = UDim2.new(0, textStart, 0, 0),
			Size = UDim2.new(1, -textStart, 1, 0),
			Font = Enum.Font.GothamBold,
			Text = config.Title,
			TextColor3 = theme.Text,
			TextSize = metrics.FontTitle + 1,
			TextXAlignment = Enum.TextXAlignment.Left,
		}, header)
	end

	local cardTab = setmetatable({
		_window = self._window,
		_page = card,
		_theme = theme,
		_metrics = metrics,
		_openRow = nil,
	}, TabMeta)

	return cardTab
end

function TabMeta:CreateButton(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local variant = config.Variant or "Primary"
	local bg, textColor, useStroke
	if variant == "Danger" then
		bg, textColor = Color3.fromRGB(217, 68, 68), Color3.new(1, 1, 1)
	elseif variant == "Outline" then
		bg, textColor, useStroke = theme.Background, theme.Accent, true
	else
		bg, textColor = theme.Accent, Color3.new(1, 1, 1)
	end

	local icon = config.Icon
	if icon == nil and variant == "Danger" then
		icon = VeloxUI.Icons.Warning
	end

	local btn = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = bg,
		Text = "",
		AutoButtonColor = false,
	}, control)
	corner(btn, metrics.Radius - 2)
	if useStroke then
		stroke(btn, theme.Accent, 1)
	end
	wireButtonFeel(btn, function() return bg end)

	local layout = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.5, 0),
		Size = UDim2.new(1, -12, 1, 0),
		BackgroundTransparency = 1,
	}, btn)
	create("UIListLayout", {
		FillDirection = Enum.FillDirection.Horizontal,
		HorizontalAlignment = Enum.HorizontalAlignment.Center,
		VerticalAlignment = Enum.VerticalAlignment.Center,
		Padding = UDim.new(0, 6),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, layout)

	if icon then
		local iconWrap = create("Frame", {
			Size = UDim2.fromOffset(14, 14),
			BackgroundTransparency = 1,
		}, layout)
		renderIcon(iconWrap, icon, 14, textColor)
	end

	create("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.new(0, 0, 1, 0),
		AutomaticSize = Enum.AutomaticSize.X,
		Font = Enum.Font.GothamBold,
		Text = config.ButtonText or "Run",
		TextColor3 = textColor,
		TextSize = metrics.FontDesc + 1,
	}, layout)

	btn.MouseButton1Click:Connect(function()
		safeCall(config.Callback)
	end)

	return {
		Instance = row,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateToggle(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local state = config.Value or false
	local switchSize = metrics.SwitchSize

	local switch = create("Frame", {
		Size = UDim2.fromOffset(switchSize.X, switchSize.Y),
		Position = UDim2.new(1, -switchSize.X, 0.5, -switchSize.Y / 2),
		BackgroundColor3 = state and theme.Accent or theme.Stroke,
	}, control)
	corner(switch, switchSize.Y / 2)

	local knobSize = switchSize.Y - 4
	local function knobPos(on)
		return on and UDim2.new(1, -knobSize - 2, 0.5, -knobSize / 2)
			or UDim2.new(0, 2, 0.5, -knobSize / 2)
	end

	local knob = create("Frame", {
		Size = UDim2.fromOffset(knobSize, knobSize),
		Position = knobPos(state),
		BackgroundColor3 = Color3.new(1, 1, 1),
	}, switch)
	corner(knob, knobSize / 2)

	local hitBox = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = "",
	}, switch)

	local function setState(newState, fire)
		state = newState
		tween(switch, { BackgroundColor3 = state and theme.Accent or theme.Stroke }, 0.15)
		tween(knob, { Position = knobPos(state) }, 0.15)
		if fire then
			safeCall(config.Callback, state)
		end
	end

	hitBox.MouseButton1Click:Connect(function()
		setState(not state, true)
	end)

	return {
		Instance = row,
		Set = function(_, v) setState(v, false) end,
		Get = function() return state end,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateSlider(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc, metrics.RowHeightTall)
	control.AnchorPoint = Vector2.new(1, 1)
	control.Position = UDim2.new(1, -12, 1, -14)
	control.Size = UDim2.new(0.92, 0, 0, 24)

	local min = config.Min or 0
	local max = config.Max or 100
	local step = config.Step or 1
	local value = config.Default or min

	local hitArea = create("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
	}, control)

	local track = create("Frame", {
		Size = UDim2.new(1, 0, 0, 6),
		Position = UDim2.new(0, 0, 0.5, -3),
		BackgroundColor3 = theme.Stroke,
	}, hitArea)
	corner(track, 3)

	local fill = create("Frame", {
		Size = UDim2.new(0, 0, 1, 0),
		BackgroundColor3 = theme.Accent,
	}, track)
	corner(fill, 3)

	local valueLabel = create("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, 0, 0, -18),
		Size = UDim2.new(0, 60, 0, 14),
		Font = Enum.Font.Gotham,
		Text = tostring(value),
		TextColor3 = theme.SubText,
		TextSize = metrics.FontDesc,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, control)

	local function apply(raw, fire)
		raw = math.clamp(raw, min, max)
		raw = math.floor((raw - min) / step + 0.5) * step + min
		raw = math.clamp(raw, min, max)
		value = raw
		fill.Size = UDim2.new((raw - min) / math.max(max - min, 1e-6), 0, 1, 0)
		valueLabel.Text = tostring(raw)
		if fire then
			safeCall(config.Callback, raw)
		end
	end
	apply(value, false)

	local function ratioFromInput(input)
		local size = track.AbsoluteSize.X
		if size <= 0 then return 0 end
		return math.clamp((input.Position.X - track.AbsolutePosition.X) / size, 0, 1)
	end

	hitArea.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			apply(min + ratioFromInput(input) * (max - min), true)
			DragBus.Begin(function(moveInput)
				apply(min + ratioFromInput(moveInput) * (max - min), true)
			end)
		end
	end)

	return {
		Instance = row,
		Set = function(_, v) apply(v, true) end,
		Get = function() return value end,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateDropdown(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local window = self._window
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local multi = config.Multi or false
	local values = config.Values or {}
	local selected = multi and (config.Value or {}) or config.Value or values[1]

	local dropBtn = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = theme.Background,
		Text = "",
		AutoButtonColor = false,
	}, control)
	corner(dropBtn, metrics.Radius - 2)
	stroke(dropBtn, theme.Stroke, 1)
	wireButtonFeel(dropBtn, function() return theme.Background end)

	local chevronWrap = create("Frame", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -8, 0.5, 0),
		Size = UDim2.fromOffset(12, 12),
	}, dropBtn)
	renderIcon(chevronWrap, VeloxUI.Icons.ChevronDown, 12, theme.SubText)

	local chipHolder, selectedLabel

	local function refreshChips()
		if not multi then return end
		for _, c in ipairs(chipHolder:GetChildren()) do
			if c:IsA("Frame") then c:Destroy() end
		end
		local shown = 0
		for _, val in ipairs(selected) do
			if shown >= 2 then break end
			local chip = create("Frame", {
				BackgroundColor3 = theme.Accent,
				BackgroundTransparency = 0.8,
				Size = UDim2.new(0, 0, 0, metrics.Compact and 26 or 20),
				AutomaticSize = Enum.AutomaticSize.X,
			}, chipHolder)
			corner(chip, 6)
			create("UIPadding", { PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6) }, chip)
			create("TextLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.new(0, 0, 1, 0),
				AutomaticSize = Enum.AutomaticSize.X,
				Font = Enum.Font.Gotham,
				Text = tostring(val),
				TextColor3 = theme.Accent,
				TextSize = metrics.FontDesc - 1,
			}, chip)
			shown = shown + 1
		end
		local extra = #selected - shown
		if extra > 0 then
			local more = create("Frame", {
				BackgroundTransparency = 1,
				Size = UDim2.new(0, 0, 0, metrics.Compact and 26 or 20),
				AutomaticSize = Enum.AutomaticSize.X,
			}, chipHolder)
			create("TextLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.new(0, 0, 1, 0),
				AutomaticSize = Enum.AutomaticSize.X,
				Font = Enum.Font.Gotham,
				Text = "+" .. extra,
				TextColor3 = theme.SubText,
				TextSize = metrics.FontDesc - 1,
			}, more)
		end
		if #selected == 0 then
			create("TextLabel", {
				BackgroundTransparency = 1,
				Size = UDim2.new(0, 0, 1, 0),
				AutomaticSize = Enum.AutomaticSize.X,
				Font = Enum.Font.Gotham,
				Text = "Select...",
				TextColor3 = theme.SubText,
				TextSize = metrics.FontDesc,
			}, chipHolder)
		end
	end

	if multi then
		chipHolder = create("Frame", {
			BackgroundTransparency = 1,
			Position = UDim2.new(0, 8, 0, 0),
			Size = UDim2.new(1, -26, 1, 0),
			ClipsDescendants = true,
		}, dropBtn)
		create("UIListLayout", {
			FillDirection = Enum.FillDirection.Horizontal,
			Padding = UDim.new(0, 4),
			VerticalAlignment = Enum.VerticalAlignment.Center,
			SortOrder = Enum.SortOrder.LayoutOrder,
		}, chipHolder)
		refreshChips()
	else
		selectedLabel = create("TextLabel", {
			BackgroundTransparency = 1,
			Position = UDim2.new(0, 10, 0, 0),
			Size = UDim2.new(1, -26, 1, 0),
			Font = Enum.Font.Gotham,
			Text = tostring(selected or "Select..."),
			TextColor3 = theme.Text,
			TextSize = metrics.FontDesc + 1,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextTruncate = Enum.TextTruncate.AtEnd,
		}, dropBtn)
	end

	local listFrame = create("Frame", {
		BackgroundColor3 = theme.Background,
		ClipsDescendants = true,
		Visible = false,
		ZIndex = 55,
	}, window._overlay)
	corner(listFrame, metrics.Radius - 2)
	stroke(listFrame, theme.Stroke, 1)
	create("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, listFrame)

	local function optionHeight()
		return metrics.Compact and 34 or 26
	end

	local function isSelected(val)
		if not multi then return selected == val end
		for _, v in ipairs(selected) do
			if v == val then return true end
		end
		return false
	end

	local function rebuild(newValues)
		values = newValues or values
		for _, c in ipairs(listFrame:GetChildren()) do
			if c:IsA("TextButton") then c:Destroy() end
		end
		for _, val in ipairs(values) do
			local opt = create("TextButton", {
				Size = UDim2.new(1, 0, 0, optionHeight()),
				BackgroundColor3 = theme.Accent,
				BackgroundTransparency = isSelected(val) and 0.85 or 1,
				Text = "",
				AutoButtonColor = false,
				ZIndex = 55,
			}, listFrame)
			local optLabel = create("TextLabel", {
				BackgroundTransparency = 1,
				Position = UDim2.new(0, 10, 0, 0),
				Size = UDim2.new(1, -28, 1, 0),
				Font = Enum.Font.Gotham,
				Text = tostring(val),
				TextColor3 = isSelected(val) and theme.Accent or theme.Text,
				TextSize = metrics.FontDesc + 1,
				TextXAlignment = Enum.TextXAlignment.Left,
				ZIndex = 55,
			}, opt)
			local check = create("Frame", {
				BackgroundTransparency = 1,
				AnchorPoint = Vector2.new(1, 0.5),
				Position = UDim2.new(1, -8, 0.5, 0),
				Size = UDim2.fromOffset(14, 14),
				Visible = isSelected(val),
				ZIndex = 55,
			}, opt)
			renderIcon(check, VeloxUI.Icons.Check, 14, theme.Accent)
			opt.MouseEnter:Connect(function()
				if not isSelected(val) then
					tween(opt, { BackgroundTransparency = 0.9 }, 0.1)
				end
			end)
			opt.MouseLeave:Connect(function()
				if not isSelected(val) then
					tween(opt, { BackgroundTransparency = 1 }, 0.1)
				end
			end)
			opt.MouseButton1Click:Connect(function()
				if multi then
					if isSelected(val) then
						for i, v in ipairs(selected) do
							if v == val then
								table.remove(selected, i)
								break
							end
						end
					else
						table.insert(selected, val)
					end
					opt.BackgroundTransparency = isSelected(val) and 0.85 or 1
					optLabel.TextColor3 = isSelected(val) and theme.Accent or theme.Text
					check.Visible = isSelected(val)
					refreshChips()
				else
					selected = val
					selectedLabel.Text = tostring(val)
					rebuild(values)
					window._closeActivePopup()
				end
				safeCall(config.Callback, selected)
			end)
		end
	end
	rebuild(values)

	dropBtn.MouseButton1Click:Connect(function()
		if listFrame.Visible then
			window._closeActivePopup()
		else
			local height = math.min(#values, 5) * optionHeight()
			local width = dropBtn.AbsoluteSize.X
			if width <= 0 then width = 160 end
			listFrame.Size = UDim2.fromOffset(width, height)
			window._openPopup(listFrame, dropBtn, width, height)
		end
	end)

	return {
		Instance = row,
		Select = function(_, v)
			selected = v
			if multi then
				refreshChips()
			elseif selectedLabel then
				selectedLabel.Text = tostring(v)
			end
		end,
		Refresh = function(_, newValues) rebuild(newValues) end,
		Get = function() return selected end,
		Destroy = function()
			if listFrame.Parent then
				listFrame:Destroy()
			end
			row:Destroy()
		end,
	}
end

function TabMeta:CreateNumber(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local min = config.Min or 0
	local max = config.Max or math.huge
	local step = config.Step or 1
	local value = config.Value or min

	local box = create("TextBox", {
		Size = UDim2.new(1, -24, 1, 0),
		BackgroundColor3 = theme.Background,
		Text = tostring(value),
		Font = Enum.Font.Gotham,
		TextColor3 = theme.Text,
		TextSize = metrics.FontDesc + 1,
		ClearTextOnFocus = false,
	}, control)
	corner(box, metrics.Radius - 2)
	local boxStroke = stroke(box, theme.Stroke, 1)
	create("UIPadding", { PaddingLeft = UDim.new(0, 8) }, box)

	local stepper = create("Frame", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, 0, 0.5, 0),
		Size = UDim2.fromOffset(20, 26),
		BackgroundTransparency = 1,
	}, control)

	local upBtn = create("TextButton", {
		Size = UDim2.new(1, 0, 0.5, -1),
		BackgroundColor3 = theme.Background,
		Text = "",
		AutoButtonColor = false,
	}, stepper)
	corner(upBtn, 4)
	renderIcon(upBtn, VeloxUI.Icons.ChevronUp, 10, theme.SubText)
	local downBtn = create("TextButton", {
		Position = UDim2.new(0, 0, 0.5, 1),
		Size = UDim2.new(1, 0, 0.5, -1),
		BackgroundColor3 = theme.Background,
		Text = "",
		AutoButtonColor = false,
	}, stepper)
	corner(downBtn, 4)
	renderIcon(downBtn, VeloxUI.Icons.ChevronDown, 10, theme.SubText)
	wireButtonFeel(upBtn, function() return theme.Background end)
	wireButtonFeel(downBtn, function() return theme.Background end)

	local function setValue(v, fire)
		v = math.clamp(v, min, max)
		value = v
		box.Text = tostring(v)
		if fire then
			safeCall(config.Callback, v)
		end
	end

	upBtn.MouseButton1Click:Connect(function() setValue(value + step, true) end)
	downBtn.MouseButton1Click:Connect(function() setValue(value - step, true) end)

	box.Focused:Connect(function()
		tween(boxStroke, { Color = theme.Accent }, 0.12)
	end)
	box.FocusLost:Connect(function()
		tween(boxStroke, { Color = theme.Stroke }, 0.12)
		local n = tonumber(box.Text)
		if n then
			setValue(n, true)
		else
			box.Text = tostring(value)
		end
	end)

	return {
		Instance = row,
		Set = function(_, v) setValue(v, false) end,
		Get = function() return value end,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateProgressBar(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)
	control.AnchorPoint = Vector2.new(1, 0.5)
	control.Position = UDim2.new(1, -12, 0.5, 0)
	control.Size = UDim2.new(0.92, 0, 0, 16)

	local min = config.Min or 0
	local max = config.Max or 100
	local value = config.Value or min

	local track = create("Frame", {
		Size = UDim2.new(1, 0, 0, 8),
		Position = UDim2.new(0, 0, 0.5, -4),
		BackgroundColor3 = theme.Stroke,
	}, control)
	corner(track, 4)

	local fill = create("Frame", {
		Size = UDim2.new(0, 0, 1, 0),
		BackgroundColor3 = theme.Accent,
	}, track)
	corner(fill, 4)

	local percentLabel = create("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, 0, 0, -4),
		Size = UDim2.new(0, 46, 0, 14),
		Font = Enum.Font.Gotham,
		TextColor3 = theme.SubText,
		TextSize = metrics.FontDesc - 1,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, control)

	local function setValue(v)
		v = math.clamp(v, min, max)
		value = v
		local ratio = (v - min) / math.max(max - min, 1e-6)
		tween(fill, { Size = UDim2.new(ratio, 0, 1, 0) }, 0.2)
		percentLabel.Text = math.floor(ratio * 100) .. "%"
	end
	setValue(value)

	return {
		Instance = row,
		Set = function(_, v) setValue(v) end,
		Get = function() return value end,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateKeybind(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local current = config.Value or "None"
	local myCaptureFn

	local btn = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = theme.Background,
		Text = "",
		AutoButtonColor = false,
	}, control)
	corner(btn, metrics.Radius - 2)
	stroke(btn, theme.Stroke, 1)
	wireButtonFeel(btn, function() return theme.Background end)

	local label = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 10, 0, 0),
		Size = UDim2.new(1, -30, 1, 0),
		Font = Enum.Font.GothamBold,
		Text = current,
		TextColor3 = theme.Text,
		TextSize = metrics.FontDesc + 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, btn)
	create("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.new(1, -8, 0.5, 0),
		Size = UDim2.fromOffset(14, 14),
		Font = Enum.Font.GothamBold,
		Text = "K",
		TextColor3 = theme.SubText,
		TextSize = 11,
	}, btn)

	btn.MouseButton1Click:Connect(function()
		label.Text = "..."
		myCaptureFn = function(keyCode)
			current = keyCode.Name
			label.Text = current
			safeCall(config.Callback, current)
		end
		InputBus.CaptureNextKey(myCaptureFn)
	end)

	return {
		Instance = row,
		Set = function(_, v) current = v; label.Text = v end,
		Get = function() return current end,
		Destroy = function()
			if myCaptureFn then
				InputBus.CancelCapture(myCaptureFn)
			end
			row:Destroy()
		end,
	}
end

function TabMeta:CreateColorpicker(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local window = self._window
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local color = config.Default or Color3.fromRGB(255, 255, 255)
	local rgb = { color.R * 255, color.G * 255, color.B * 255 }

	local preview = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = color,
		Text = "",
		AutoButtonColor = false,
	}, control)
	corner(preview, metrics.Radius - 2)
	stroke(preview, theme.Stroke, 1)

	local popupWidth = 200
	local popupHeight = 130

	local popup = create("Frame", {
		BackgroundColor3 = theme.Background,
		Visible = false,
		ZIndex = 55,
	}, window._overlay)
	corner(popup, metrics.Radius)
	stroke(popup, theme.Stroke, 1)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 10), PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 10),
	}, popup)
	create("UIListLayout", { Padding = UDim.new(0, 8) }, popup)

	local function updateColor()
		color = Color3.fromRGB(
			math.clamp(rgb[1], 0, 255),
			math.clamp(rgb[2], 0, 255),
			math.clamp(rgb[3], 0, 255)
		)
		preview.BackgroundColor3 = color
		safeCall(config.Callback, color)
	end

	local function makeChannel(name, index)
		local holder = create("Frame", {
			Size = UDim2.new(1, 0, 0, 26), BackgroundTransparency = 1, ZIndex = 55,
		}, popup)
		create("TextLabel", {
			BackgroundTransparency = 1,
			Size = UDim2.new(0, 16, 1, 0),
			Text = name,
			Font = Enum.Font.GothamBold,
			TextColor3 = theme.Text,
			TextSize = metrics.FontDesc,
			ZIndex = 55,
		}, holder)
		local track = create("Frame", {
			Position = UDim2.new(0, 20, 0.5, -3),
			Size = UDim2.new(1, -20, 0, 6),
			BackgroundColor3 = theme.Stroke,
			ZIndex = 55,
		}, holder)
		corner(track, 3)
		local fill = create("Frame", {
			Size = UDim2.new(rgb[index] / 255, 0, 1, 0),
			BackgroundColor3 = theme.Accent,
			ZIndex = 55,
		}, track)
		corner(fill, 3)

		local function ratioFromInput(input)
			local size = track.AbsoluteSize.X
			if size <= 0 then return 0 end
			return math.clamp((input.Position.X - track.AbsolutePosition.X) / size, 0, 1)
		end

		local function set(ratio)
			fill.Size = UDim2.new(ratio, 0, 1, 0)
			rgb[index] = ratio * 255
			updateColor()
		end

		holder.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				set(ratioFromInput(input))
				DragBus.Begin(function(moveInput)
					set(ratioFromInput(moveInput))
				end)
			end
		end)
	end

	makeChannel("R", 1)
	makeChannel("G", 2)
	makeChannel("B", 3)

	preview.MouseButton1Click:Connect(function()
		if popup.Visible then
			window._closeActivePopup()
		else
			local vp = Viewport.Get()
			local width = math.min(popupWidth, vp.X - 24)
			popup.Size = UDim2.fromOffset(width, popupHeight)
			window._openPopup(popup, preview, width, popupHeight)
		end
	end)

	return {
		Instance = row,
		Get = function() return color end,
		Destroy = function()
			if popup.Parent then
				popup:Destroy()
			end
			row:Destroy()
		end,
	}
end

return VeloxUI
