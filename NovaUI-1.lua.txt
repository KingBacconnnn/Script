--[[
	VeloxUI Official — a modern, responsive, single-file UI library for Roblox

	A ground-up refresh of the previous NovaUI library: same general shape
	(Window > Tab > controls), rebuilt to behave well on phones, tablets,
	and desktop, with far fewer persistent connections than before.

	What changed in this pass:
		• Responsive window sizing plus a collapsible mobile sidebar drawer
		• Automatic adaptation to screen size, orientation, and touch input
		• Dropdowns / colorpickers now open in a screen-space overlay layer
		  and are clamped so they can never render off-screen
		• All dragging (window move, sliders, colorpicker channels) shares
		  two library-wide input connections instead of creating new ones
		  per control — connection count no longer grows with UI size
		• Keybind capture and ToggleKey shortcuts share one input listener
		• Hover / pressed / focused / selected states, smoother tweens, and
		  a more consistent visual language across every control
		• Window / Tab / control Destroy() now unwinds everything it
		  registered, including shared-service subscriptions

	The public API is unchanged from the previous version — CreateWindow,
	CreateTab, CreateButton, CreateToggle, CreateSlider, CreateDropdown,
	CreateInput, CreateKeybind, CreateColorpicker, CreateParagraph,
	CreateDivider, CreateSection, and Notify all take the same fields as
	before, so existing scripts should keep working with, at most, a
	rename of the variable you loadstring this into.

	Basic usage:

		local VeloxUI = loadstring(game:HttpGet("RAW_LINK_TO_THIS_FILE"))()

		local Window = VeloxUI:CreateWindow({ Title = "My Hub" })
		local Tab = Window:CreateTab({ Title = "Main" })

		Tab:CreateButton({
			Title = "Say Hi",
			Callback = function() print("hi!") end,
		})
]]

local function create(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props or {}) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function corner(inst, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius or 8)
	c.Parent = inst
	return c
end

local function stroke(inst, color, thickness)
	local s = Instance.new("UIStroke")
	s.Color = color or Color3.new(1, 1, 1)
	s.Thickness = thickness or 1
	s.Parent = inst
	return s
end

-- Reused TweenInfos so we're not allocating a new one on every call.
local TWEEN_FAST   = TweenInfo.new(0.10, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local TWEEN_NORMAL = TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local TWEEN_SLOW   = TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function tween(inst, props, time)
	local info = TWEEN_NORMAL
	if time then
		if time <= 0.12 then
			info = TWEEN_FAST
		elseif time >= 0.25 then
			info = TWEEN_SLOW
		else
			info = TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
		end
	end
	local t = TweenService:Create(inst, info, props)
	t:Play()
	return t
end

-- Nudges a color's brightness up or down (HSV space) for hover / pressed
-- feedback, so we don't need extra colors in every theme table.
local function shade(color, delta)
	local h, s, v = color:ToHSV()
	v = math.clamp(v + delta, 0, 1)
	return Color3.fromHSV(h, s, v)
end

local function safeCall(fn, ...)
	if not fn then return end
	local ok, err = pcall(fn, ...)
	if not ok then
		warn("[VeloxUI] Callback error: " .. tostring(err))
	end
	return ok
end

-- Hover / pressed color feedback for a button-like GuiObject. getBase()
-- returns the control's current resting color, since some controls
-- (toggles, dropdowns) change their base color over time.
local function wireButtonFeel(btn, getBase)
	local function reset()
		tween(btn, { BackgroundColor3 = getBase() }, 0.12)
	end
	btn.MouseEnter:Connect(function()
		tween(btn, { BackgroundColor3 = shade(getBase(), 0.08) }, 0.12)
	end)
	btn.MouseLeave:Connect(reset)
	btn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			tween(btn, { BackgroundColor3 = shade(getBase(), -0.08) }, 0.08)
		end
	end)
	btn.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			reset()
		end
	end)
end

----------------------------------------------------------------
-- Viewport — screen size / device type, one shared listener set
-- no matter how many windows subscribe to it.
----------------------------------------------------------------

local Viewport = {}
do
	local listeners = {}

	local function getCamera()
		return Workspace.CurrentCamera
	end

	function Viewport.Get()
		local cam = getCamera()
		if cam then
			return cam.ViewportSize
		end
		return Vector2.new(1280, 720)
	end

	function Viewport.IsCompact()
		local size = Viewport.Get()
		local touchOnly = UserInputService.TouchEnabled and not UserInputService.MouseEnabled
		return touchOnly or size.X < 650
	end

	local function fire()
		for _, fn in ipairs(listeners) do
			pcall(fn)
		end
	end

	function Viewport.OnChange(fn)
		table.insert(listeners, fn)
		return function()
			local idx = table.find(listeners, fn)
			if idx then
				table.remove(listeners, idx)
			end
		end
	end

	local function hook(cam)
		if cam then
			cam:GetPropertyChangedSignal("ViewportSize"):Connect(fire)
		end
	end

	hook(getCamera())
	Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
		hook(getCamera())
		fire()
	end)
end

----------------------------------------------------------------
-- InputBus — a single keyboard listener shared by every ToggleKey
-- and every Keybind control, instead of one connection each.
----------------------------------------------------------------

local InputBus = {}
do
	local captureCallback
	local toggleBindings = {}

	UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if input.UserInputType ~= Enum.UserInputType.Keyboard then
			return
		end

		if captureCallback then
			local cb = captureCallback
			captureCallback = nil
			cb(input.KeyCode)
			return
		end

		if gameProcessed then
			return
		end

		for _, binding in ipairs(toggleBindings) do
			if binding.keycode == input.KeyCode then
				safeCall(binding.fn)
			end
		end
	end)

	-- Calls callback(keyCode) the next time any key is pressed, then
	-- stops listening automatically.
	function InputBus.CaptureNextKey(callback)
		captureCallback = callback
	end

	function InputBus.CancelCapture(callback)
		if captureCallback == callback then
			captureCallback = nil
		end
	end

	-- Returns an unregister function.
	function InputBus.RegisterToggleKey(keycode, fn)
		local entry = { keycode = keycode, fn = fn }
		table.insert(toggleBindings, entry)
		return function()
			local idx = table.find(toggleBindings, entry)
			if idx then
				table.remove(toggleBindings, idx)
			end
		end
	end
end

----------------------------------------------------------------
-- DragBus — shared pointer tracking for window dragging, sliders,
-- and colorpicker channels. Exactly two permanent connections for
-- the whole library, no matter how many draggable controls exist.
----------------------------------------------------------------

local DragBus = {}
do
	local activeMove, activeEnd

	UserInputService.InputChanged:Connect(function(input)
		if activeMove and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			activeMove(input)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if activeEnd and (input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch) then
			local fn = activeEnd
			activeMove, activeEnd = nil, nil
			fn(input)
		end
	end)

	function DragBus.Begin(onMove, onEnd)
		activeMove = onMove
		activeEnd = onEnd or function() end
	end

	function DragBus.Cancel()
		activeMove, activeEnd = nil, nil
	end
end

local function makeDraggable(handle, target)
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			local startPos = target.Position
			local startInput = input.Position
			DragBus.Begin(function(moveInput)
				local delta = moveInput.Position - startInput
				target.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + delta.X,
					startPos.Y.Scale, startPos.Y.Offset + delta.Y
				)
			end)
		end
	end)
end

local NotifHolder

local function ensureNotifHolder()
	if NotifHolder and NotifHolder.Parent then
		return NotifHolder
	end

	local gui = create("ScreenGui", {
		Name = "VeloxUI_Notifications",
		ResetOnSpawn = false,
		IgnoreGuiInset = true,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	})

	local ok = pcall(function()
		gui.Parent = game:GetService("CoreGui")
	end)
	if not ok then
		gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
	end

	local function holderSize()
		local vp = Viewport.Get()
		return UDim2.new(0, math.min(300, vp.X - 32), 1, -32)
	end

	NotifHolder = create("Frame", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, -16, 1, -16),
		Size = holderSize(),
	}, gui)

	create("UIListLayout", {
		Padding = UDim.new(0, 8),
		VerticalAlignment = Enum.VerticalAlignment.Bottom,
		HorizontalAlignment = Enum.HorizontalAlignment.Right,
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, NotifHolder)

	Viewport.OnChange(function()
		if NotifHolder and NotifHolder.Parent then
			NotifHolder.Size = holderSize()
		end
	end)

	return NotifHolder
end

function VeloxUI:Notify(config)
	config = config or {}
	local theme = self.CurrentTheme
	local holder = ensureNotifHolder()

	local frame = create("Frame", {
		BackgroundColor3 = theme.Element,
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		ClipsDescendants = true,
	}, holder)
	corner(frame, 10)
	stroke(frame, theme.Stroke, 1)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10),
		PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 12),
	}, frame)

	local accentBar = create("Frame", {
		Size = UDim2.new(0, 3, 1, -12),
		Position = UDim2.new(0, 0, 0, 6),
		BackgroundColor3 = theme.Accent,
		BackgroundTransparency = 1,
	}, frame)
	corner(accentBar, 2)

	local title = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 8, 0, 0),
		Size = UDim2.new(1, -8, 0, 18),
		Font = Enum.Font.GothamBold,
		Text = config.Title or "Notification",
		TextColor3 = theme.Text,
		TextTransparency = 1,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, frame)

	local content = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 8, 0, 20),
		Size = UDim2.new(1, -8, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		Font = Enum.Font.Gotham,
		Text = config.Content or "",
		TextColor3 = theme.SubText,
		TextTransparency = 1,
		TextSize = 13,
		TextWrapped = true,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, frame)

	tween(frame, { BackgroundTransparency = 0 }, 0.18)
	tween(accentBar, { BackgroundTransparency = 0 }, 0.18)
	tween(title, { TextTransparency = 0 }, 0.18)
	tween(content, { TextTransparency = 0 }, 0.18)

	task.delay(config.Duration or 3, function()
		if not (frame and frame.Parent) then return end
		tween(frame, { BackgroundTransparency = 1 }, 0.2)
		tween(accentBar, { BackgroundTransparency = 1 }, 0.2)
		tween(title, { TextTransparency = 1 }, 0.2)
		local hideTween = tween(content, { TextTransparency = 1 }, 0.2)
		hideTween.Completed:Connect(function()
			if frame and frame.Parent then
				frame:Destroy()
			end
		end)
	end)
end

local function buildMetrics(compact)
	if compact then
		return {
			Compact = true,
			RowHeight = 54,
			RowHeightTall = 64,
			FontTitle = 15,
			FontDesc = 13,
			ControlWidth = 0.46,
			Radius = 10,
			SwitchSize = Vector2.new(46, 26),
			TopBarHeight = 46,
			SidebarWidth = 190,
		}
	else
		return {
			Compact = false,
			RowHeight = 44,
			RowHeightTall = 52,
			FontTitle = 14,
			FontDesc = 12,
			ControlWidth = 0.4,
			Radius = 8,
			SwitchSize = Vector2.new(40, 22),
			TopBarHeight = 42,
			SidebarWidth = 150,
		}
	end
end

local WindowMeta = {}
WindowMeta.__index = WindowMeta

local function defaultWindowSize(compact)
	local vp = Viewport.Get()
	if compact then
		local w = math.clamp(vp.X - 24, 220, 460)
		local h = math.clamp(vp.Y - 80, 200, 620)
		return UDim2.fromOffset(math.floor(w), math.floor(h))
	else
		local w = math.clamp(vp.X * 0.5, 480, 640)
		local h = math.clamp(vp.Y * 0.55, 360, 460)
		return UDim2.fromOffset(math.floor(w), math.floor(h))
	end
end

function VeloxUI:CreateWindow(config)
	config = config or {}
	local theme = (config.Theme and self.Themes[config.Theme]) or self.CurrentTheme
	self.CurrentTheme = theme

	local compact = Viewport.IsCompact()
	local metrics = buildMetrics(compact)
	local usesCustomSize = config.Size ~= nil
	local sidebarWidth = config.SidebarWidth or metrics.SidebarWidth
	local topBarHeight = metrics.TopBarHeight

	local gui = create("ScreenGui", {
		Name = config.Folder or "VeloxUI",
		ResetOnSpawn = false,
		IgnoreGuiInset = true,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	})
	local guiOk = pcall(function()
		gui.Parent = game:GetService("CoreGui")
	end)
	if not guiOk then
		gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
	end

	local windowSize = usesCustomSize and config.Size or defaultWindowSize(compact)

	local function centeredPosition(size)
		local vp = Viewport.Get()
		local x = math.max(math.floor((vp.X - size.X.Offset) / 2), 0)
		local y = math.max(math.floor((vp.Y - size.Y.Offset) / 2), 0)
		return UDim2.fromOffset(x, y)
	end

	local main = create("Frame", {
		Size = windowSize,
		Position = centeredPosition(windowSize),
		BackgroundColor3 = theme.Background,
		ClipsDescendants = true,
	}, gui)
	corner(main, metrics.Radius + 2)
	stroke(main, theme.Stroke, 1)

	local topBar = create("Frame", {
		Size = UDim2.new(1, 0, 0, topBarHeight),
		BackgroundColor3 = theme.Sidebar,
		BorderSizePixel = 0,
	}, main)
	corner(topBar, metrics.Radius + 2)
	create("Frame", {
		Size = UDim2.new(1, 0, 0, metrics.Radius + 2),
		Position = UDim2.new(0, 0, 1, -(metrics.Radius + 2)),
		BackgroundColor3 = theme.Sidebar,
		BorderSizePixel = 0,
		ZIndex = 0,
	}, topBar)

	local hamburger = create("TextButton", {
		Size = UDim2.fromOffset(topBarHeight - 16, topBarHeight - 16),
		Position = UDim2.new(0, 8, 0.5, -(topBarHeight - 16) / 2),
		BackgroundColor3 = theme.Element,
		Text = "=",
		Font = Enum.Font.GothamBold,
		TextColor3 = theme.Text,
		TextSize = 16,
		AutoButtonColor = false,
		Visible = false,
	}, topBar)
	corner(hamburger, 6)
	wireButtonFeel(hamburger, function() return theme.Element end)

	local title = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 16, 0, 0),
		Size = UDim2.new(1, -110, 1, 0),
		Font = Enum.Font.GothamBold,
		Text = config.Title or "VeloxUI Official",
		TextColor3 = theme.Text,
		TextSize = 16,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, topBar)

	local closeBtn = create("TextButton", {
		Size = UDim2.fromOffset(28, 28),
		Position = UDim2.new(1, -36, 0.5, -14),
		BackgroundColor3 = theme.Element,
		Text = "X",
		Font = Enum.Font.GothamBold,
		TextColor3 = theme.Text,
		TextSize = 12,
		AutoButtonColor = false,
	}, topBar)
	corner(closeBtn, 6)
	wireButtonFeel(closeBtn, function() return theme.Element end)

	local minBtn = create("TextButton", {
		Size = UDim2.fromOffset(28, 28),
		Position = UDim2.new(1, -70, 0.5, -14),
		BackgroundColor3 = theme.Element,
		Text = "_",
		Font = Enum.Font.GothamBold,
		TextColor3 = theme.Text,
		TextSize = 14,
		AutoButtonColor = false,
	}, topBar)
	corner(minBtn, 6)
	wireButtonFeel(minBtn, function() return theme.Element end)

	makeDraggable(topBar, main)

	local sidebar = create("Frame", {
		BackgroundColor3 = theme.Sidebar,
		BorderSizePixel = 0,
	}, main)

	local backdrop = create("TextButton", {
		Size = UDim2.new(1, 0, 1, -topBarHeight),
		Position = UDim2.new(0, 0, 0, topBarHeight),
		BackgroundColor3 = Color3.new(0, 0, 0),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		Visible = false,
		ZIndex = 15,
	}, main)

	local tabList = create("ScrollingFrame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 3,
		CanvasSize = UDim2.new(0, 0, 0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		ZIndex = 20,
	}, sidebar)
	create("UIListLayout", {
		Padding = UDim.new(0, 4),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, tabList)
	create("UIPadding", {
		PaddingTop = UDim.new(0, 8), PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8),
	}, tabList)

	local content = create("Frame", {
		BackgroundTransparency = 1,
	}, main)

	local overlay = create("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		ZIndex = 50,
	}, gui)

	local popupCatcher = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = "",
		AutoButtonColor = false,
		Visible = false,
		ZIndex = 52,
	}, overlay)

	local activePopup = nil

	local function closeActivePopup()
		if activePopup then
			activePopup.Visible = false
			activePopup = nil
		end
		popupCatcher.Visible = false
	end

	local function openPopup(popupFrame, anchor, width, height)
		if activePopup and activePopup ~= popupFrame then
			activePopup.Visible = false
		end
		local vp = Viewport.Get()
		local anchorPos = anchor.AbsolutePosition
		local anchorSize = anchor.AbsoluteSize
		local x = anchorPos.X
		local y = anchorPos.Y + anchorSize.Y + 4
		if x + width > vp.X then
			x = math.max(vp.X - width - 8, 0)
		end
		if y + height > vp.Y then
			y = anchorPos.Y - height - 4
			if y < 0 then
				y = math.max(vp.Y - height - 8, 0)
			end
		end
		popupFrame.Position = UDim2.fromOffset(x, y)
		popupFrame.Visible = true
		popupCatcher.Visible = true
		activePopup = popupFrame
	end

	popupCatcher.MouseButton1Click:Connect(closeActivePopup)

	local drawerOpen = false

	local function applyLayoutMode()
		if compact then
			local width = math.min(sidebarWidth, Viewport.Get().X - 40)
			sidebar.Position = UDim2.new(0, drawerOpen and 0 or -sidebarWidth, 0, topBarHeight)
			sidebar.Size = UDim2.new(0, width, 1, -topBarHeight)
			sidebar.ZIndex = 20
			content.Position = UDim2.new(0, 0, 0, topBarHeight)
			content.Size = UDim2.new(1, 0, 1, -topBarHeight)
			hamburger.Visible = true
			title.Position = UDim2.new(0, topBarHeight + 4, 0, 0)
			title.Size = UDim2.new(1, -(topBarHeight + 78), 1, 0)
		else
			sidebar.Position = UDim2.new(0, 0, 0, topBarHeight)
			sidebar.Size = UDim2.new(0, sidebarWidth, 1, -topBarHeight)
			sidebar.ZIndex = 1
			content.Position = UDim2.new(0, sidebarWidth, 0, topBarHeight)
			content.Size = UDim2.new(1, -sidebarWidth, 1, -topBarHeight)
			hamburger.Visible = false
			drawerOpen = false
			backdrop.Visible = false
			title.Position = UDim2.new(0, 16, 0, 0)
			title.Size = UDim2.new(1, -110, 1, 0)
		end
	end

	local function setDrawerOpen(open)
		if not compact then return end
		drawerOpen = open
		tween(sidebar, {
			Position = UDim2.new(0, open and 0 or -sidebarWidth, 0, topBarHeight),
		}, 0.2)
		if open then
			backdrop.Visible = true
			backdrop.BackgroundTransparency = 1
			tween(backdrop, { BackgroundTransparency = 0.45 }, 0.2)
		else
			local hideTween = tween(backdrop, { BackgroundTransparency = 1 }, 0.2)
			hideTween.Completed:Connect(function()
				if not drawerOpen then
					backdrop.Visible = false
				end
			end)
		end
	end

	hamburger.MouseButton1Click:Connect(function()
		setDrawerOpen(not drawerOpen)
	end)
	backdrop.MouseButton1Click:Connect(function()
		setDrawerOpen(false)
	end)

	applyLayoutMode()

	closeBtn.MouseButton1Click:Connect(function()
		gui:Destroy()
	end)

	local minimized = false
	local fullSize = main.Size
	minBtn.MouseButton1Click:Connect(function()
		minimized = not minimized
		if minimized then
			tween(main, { Size = UDim2.new(fullSize.X.Scale, fullSize.X.Offset, 0, topBarHeight) }, 0.22)
		else
			tween(main, { Size = fullSize }, 0.22)
		end
	end)

	local function clampMainPosition()
		local vp = Viewport.Get()
		local size = main.Size
		local pos = main.Position
		local maxX = math.max(vp.X - size.X.Offset, 0)
		local maxY = math.max(vp.Y - size.Y.Offset, 0)
		local x = math.clamp(pos.X.Offset, 0, maxX)
		local y = math.clamp(pos.Y.Offset, 0, maxY)
		if x ~= pos.X.Offset or y ~= pos.Y.Offset then
			main.Position = UDim2.new(pos.X.Scale, x, pos.Y.Scale, y)
		end
	end

	local unregisterToggleKey
	if config.ToggleKey then
		unregisterToggleKey = InputBus.RegisterToggleKey(config.ToggleKey, function()
			main.Visible = not main.Visible
		end)
	end

	local unsubscribeViewport = Viewport.OnChange(function()
		local wasCompact = compact
		compact = Viewport.IsCompact()
		if compact ~= wasCompact then
			applyLayoutMode()
		end
		if not usesCustomSize and not minimized then
			fullSize = defaultWindowSize(compact)
			main.Size = fullSize
		end
		clampMainPosition()
	end)

	local window = setmetatable({
		_theme = theme,
		_metrics = metrics,
		_gui = gui,
		_main = main,
		_tabList = tabList,
		_content = content,
		_tabs = {},
		_overlay = overlay,
		_openPopup = openPopup,
		_closeActivePopup = closeActivePopup,
		_setDrawerOpen = setDrawerOpen,
		_unsubscribeViewport = unsubscribeViewport,
		_unregisterToggleKey = unregisterToggleKey,
	}, WindowMeta)

	return window
end

function WindowMeta:Toggle()
	self._main.Visible = not self._main.Visible
end

function WindowMeta:Destroy()
	if self._unsubscribeViewport then
		self._unsubscribeViewport()
	end
	if self._unregisterToggleKey then
		self._unregisterToggleKey()
	end
	self._closeActivePopup()
	self._gui:Destroy()
end

function TabMeta:Select()
	local window = self._window
	for _, t in ipairs(window._tabs) do
		t._selected = false
		t._page.Visible = false
		tween(t._button, { BackgroundTransparency = 1 }, 0.15)
		tween(t._indicator, { BackgroundTransparency = 1 }, 0.15)
		tween(t._label, { TextColor3 = t._theme.SubText }, 0.15)
	end

	self._selected = true
	self._page.Visible = true
	tween(self._button, { BackgroundTransparency = 0 }, 0.15)
	tween(self._indicator, { BackgroundTransparency = 0 }, 0.15)
	tween(self._label, { TextColor3 = self._theme.Text }, 0.15)

	if window._setDrawerOpen then
		window._setDrawerOpen(false)
	end
end

function TabMeta:Destroy()
	local window = self._window
	local idx = table.find(window._tabs, self)
	if idx then
		table.remove(window._tabs, idx)
	end
	self._button:Destroy()
	self._page:Destroy()
	if self._selected and #window._tabs > 0 then
		window._tabs[1]:Select()
	end
end

function TabMeta:CreateToggle(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local state = config.Value or false
	local switchSize = metrics.SwitchSize

	local switch = create("Frame", {
		Size = UDim2.fromOffset(switchSize.X, switchSize.Y),
		Position = UDim2.new(1, -switchSize.X, 0.5, -switchSize.Y / 2),
		BackgroundColor3 = state and theme.Accent or theme.Stroke,
	}, control)
	corner(switch, switchSize.Y / 2)

	local knobSize = switchSize.Y - 4
	local function knobPos(on)
		return on and UDim2.new(1, -knobSize - 2, 0.5, -knobSize / 2)
			or UDim2.new(0, 2, 0.5, -knobSize / 2)
	end

	local knob = create("Frame", {
		Size = UDim2.fromOffset(knobSize, knobSize),
		Position = knobPos(state),
		BackgroundColor3 = Color3.new(1, 1, 1),
	}, switch)
	corner(knob, knobSize / 2)

	local hitBox = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
		Text = "",
	}, switch)

	local function setState(newState, fire)
		state = newState
		tween(switch, { BackgroundColor3 = state and theme.Accent or theme.Stroke }, 0.15)
		tween(knob, { Position = knobPos(state) }, 0.15)
		if fire then
			safeCall(config.Callback, state)
		end
	end

	hitBox.MouseButton1Click:Connect(function()
		setState(not state, true)
	end)

	return {
		Instance = row,
		Set = function(_, v) setState(v, false) end,
		Get = function() return state end,
		Destroy = function() row:Destroy() end,
	}
end

function TabMeta:CreateSlider(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc, metrics.RowHeightTall)
	control.AnchorPoint = Vector2.new(1, 1)
	control.Position = UDim2.new(1, -12, 1, -14)
	control.Size = UDim2.new(0.92, 0, 0, 24)

	local min = config.Min or 0
	local max = config.Max or 100
	local step = config.Step or 1
	local value = config.Default or min

	-- Slightly taller than the visible track so touch drags have a
	-- comfortable hit area without changing how the slider looks.
	local hitArea = create("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundTransparency = 1,
	}, control)

	local track = create("Frame", {
		Size = UDim2.new(1, 0, 0, 6),
		Position = UDim2.new(0, 0, 0.5, -3),
		BackgroundColor3 = theme.Stroke,
	}, hitArea)
	corner(track, 3)

	local fill = create("Frame", {
		Size = UDim2.new(0, 0, 1, 0),
		BackgroundColor3 = theme.Accent,
	}, track)
	corner(fill, 3)

	local valueLabel = create("TextLabel", {
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, 0, 0, -18),
		Size = UDim2.new(0, 60, 0, 14),
		Font = Enum.Font.Gotham,
		Text = tostring(value),
		TextColor3 = theme.SubText,
		TextSize = metrics.FontDesc,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, control)

	local function apply(raw, fire)
		raw = math.clamp(raw, min, max)
		raw = math.floor((raw - min) / step + 0.5) * step + min
		raw = math.clamp(raw, min, max)
		value = raw
		fill.Size = UDim2.new((raw - min) / math.max(max - min, 1e-6), 0, 1, 0)
		valueLabel.Text = tostring(raw)
		if fire then
			safeCall(config.Callback, raw)
		end
	end
	apply(value, false)

	local function ratioFromInput(input)
		return math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
	end

	hitArea.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			apply(min + ratioFromInput(input) * (max - min), true)
			DragBus.Begin(function(moveInput)
				apply(min + ratioFromInput(moveInput) * (max - min), true)
			end)
		end
	end)

	return {
		Instance = row,
		Set = function(_, v) apply(v, true) end,
		Get = function() return value end,
		Destroy = function() row:Destroy() end,
	}
end

-- Dropdown (opens in the window's overlay layer, clamped on-screen)
----------------------------------------------------------------

function TabMeta:CreateDropdown(config)
	config = config or {}
	local theme, metrics = self._theme, self._metrics
	local window = self._window
	local row, control = baseRow(self._page, theme, metrics, config.Title, config.Desc)

	local values = config.Values or {}
	local selected = config.Value or values[1]

	local dropBtn = create("TextButton", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = theme.Background,
		Text = "",
		AutoButtonColor = false,
	}, control)
	corner(dropBtn, metrics.Radius - 2)
	stroke(dropBtn, theme.Stroke, 1)
	wireButtonFeel(dropBtn, function() return theme.Background end)

	local selectedLabel = create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 10, 0, 0),
		Size = UDim2.new(1, -20, 1, 0),
		Font = Enum.Font.Gotham,
		Text = tostring(selected or "Select..."),
		TextColor3 = theme.Text,
		TextSize = metrics.FontDesc + 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
	}, dropBtn)

	local listFrame = create("Frame", {
		BackgroundColor3 = theme.Background,
		ClipsDescendants = true,
		Visible = false,
		ZIndex = 55,
	}, window._overlay)
	corner(listFrame, metrics.Radius - 2)
	stroke(listFrame, theme.Stroke, 1)
	create("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, listFrame)

	local function optionHeight()
		return metrics.Compact and 34 or 26
	end

	local function rebuild(newValues)
		values = newValues or values
		for _, c in ipairs(listFrame:GetChildren()) do
			if c:IsA("TextButton") then c:Destroy() end
		end
		for _, val in ipairs(values) do
			local opt = create("TextButton", {
				Size = UDim2.new(1, 0, 0, optionHeight()),
				BackgroundColor3 = theme.Element,
				BackgroundTransparency = 1,
				Text = tostring(val),
				Font = Enum.Font.Gotham,
				TextColor3 = theme.Text,
				TextSize = metrics.FontDesc + 1,
				ZIndex = 55,
			}, listFrame)
			opt.MouseEnter:Connect(function()
				tween(opt, { BackgroundTransparency = 0.85 }, 0.1)
			end)
			opt.MouseLeave:Connect(function()
				tween(opt, { BackgroundTransparency = 1 }, 0.1)
			end)
			opt.MouseButton1Click:Connect(function()
				selected = val
				selectedLabel.Text = tostring(val)
				window._closeActivePopup()
				safeCall(config.Callback, val)
			end)
		end
	end
	rebuild(values)

	dropBtn.MouseButton1Click:Connect(function()
		if listFrame.Visible then
			window._closeActivePopup()
		else
			local height = math.min(#values, 5) * optionHeight()
			local width = dropBtn.AbsoluteSize.X
			listFrame.Size = UDim2.fromOffset(width, height)
			window._openPopup(listFrame, dropBtn, width, height)
		end
	end)

	return {
		Instance = row,
		Select = function(_, v) selected = v; selectedLabel.Text = tostring(v) end,
		Refresh = function(_, newValues) rebuild(newValues) end,
		Get = function() return selected end,
		Destroy = function()
			if listFrame.Parent then
				listFrame:Destroy()
			end
			row:Destroy()
		end,
	}
end

			Position = UDim2.new(0, 20, 0.5, -3),
			Size = UDim2.new(1, -20, 0, 6),
			BackgroundColor3 = theme.Stroke,
			ZIndex = 55,
		}, holder)
		corner(track, 3)
		local fill = create("Frame", {
			Size = UDim2.new(rgb[index] / 255, 0, 1, 0),
			BackgroundColor3 = theme.Accent,
			ZIndex = 55,
		}, track)
		corner(fill, 3)

		local function ratioFromInput(input)
			return math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		end

		local function set(ratio)
			fill.Size = UDim2.new(ratio, 0, 1, 0)
			rgb[index] = ratio * 255
			updateColor()
		end

		holder.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				set(ratioFromInput(input))
				DragBus.Begin(function(moveInput)
					set(ratioFromInput(moveInput))
				end)
			end
		end)
	end

	makeChannel("R", 1)
	makeChannel("G", 2)
	makeChannel("B", 3)

	preview.MouseButton1Click:Connect(function()
		if popup.Visible then
			window._closeActivePopup()
		else
			local vp = Viewport.Get()
			local width = math.min(popupWidth, vp.X - 24)
			popup.Size = UDim2.fromOffset(width, popupHeight)
			window._openPopup(popup, preview, width, popupHeight)
		end
	end)

	return {
		Instance = row,
		Get = function() return color end,
		Destroy = function()
			if popup.Parent then
				popup:Destroy()
			end
			row:Destroy()
		end,
	}
end

return VeloxUI
