local suffixes = {
    ["K"] = 3, ["M"] = 6, ["B"] = 9, ["T"] = 12,
    ["QA"] = 15, ["QI"] = 18, ["SX"] = 21, ["SP"] = 24,
    ["OC"] = 27, ["NO"] = 30, ["DC"] = 33, ["UD"] = 36,
    ["DD"] = 39, ["TD"] = 42, ["QT"] = 45, ["QN"] = 48,
    ["QD"] = 51, ["SD"] = 54, ["OD"] = 57, ["ND"] = 60,
    ["VG"] = 63
}

local function parseMoneyString(str)
    if type(str) == "number" then return str end
    if type(str) ~= "string" then return nil end

    str = str:gsub(",", "")
    str = str:gsub("%$", "")
    str = str:gsub("%s+", "")

    local plain = tonumber(str)
    if plain then return plain end

    local num, suffix = str:match("^([%d%.]+)([%a]+)$")
    if num and suffix then
        local base = tonumber(num)
        if base then
            local power = suffixes[string.upper(suffix)]
            if power then
                return base * (10 ^ power)
            end
        end
    end

    local sciBase, sciExp = str:match("^([%d%.]+)e%+?(%d+)$")
    if sciBase and sciExp then
        local base = tonumber(sciBase)
        local exp = tonumber(sciExp)
        if base and exp then
            return base * (10 ^ exp)
        end
    end

    return nil
end

local function craftRelic(relicName)
    if isCrafting then return end
    local id = relicIdByName[relicName]
    if not id then return end
    local cost = relicCostById[id]
    if Essence < cost then return end
    isCrafting = true
    pcall(function() StartCraftRemote:InvokeServer(id) end)
    task.wait(0.5)
    isCrafting = false
    refreshEssence()
end

    end
end)

local function getRelicScore(relic)
    local tier = relic.Tier or 0
    local buffs = relic.Buffs or {}
    local score = tier * 1000000
    for _, v in pairs(buffs) do score = score + (tonumber(v) or 0) end
    return score
end

local function autoEquipBestRelic()
    if not autoEquipRelicEnabled or isEquippingRelic then return end
    isEquippingRelic = true
    pcall(function()
        RelicsService:GetState():andThen(function(state)
            if type(state) ~= "table" then
                isEquippingRelic = false
                return
            end

            local owned = state.OwnedRelics or {}
            local equipped = {}
            if type(state.EquippedRelicIds) == "table" then
                for _, id in ipairs(state.EquippedRelicIds) do
                    equipped[tonumber(id)] = true
                end
            end

            local ownedList = {}
            for id in pairs(owned) do
                local nid = tonumber(id)
                if nid and RelicsConfig.ItemsById[nid] then
                    table.insert(ownedList, nid)
                end
            end
            table.sort(ownedList, function(a, b)
                return getRelicScore(RelicsConfig.ItemsById[a]) > getRelicScore(RelicsConfig.ItemsById[b])
            end)

            local rebirthLevel = tonumber(LocalPlayer:GetAttribute("RebirthLevel")) or 0
            local maxSlots = 2
            if rebirthLevel >= 5 then maxSlots = maxSlots + 1 end
            if rebirthLevel >= 12 then maxSlots = maxSlots + 1 end
            if state.RobuxSlotUnlocked == true then maxSlots = maxSlots + 1 end

            local equippedCount = 0
            for _ in pairs(equipped) do equippedCount = equippedCount + 1 end

            local toEquip = {}
            for _, id in ipairs(ownedList) do
                if not equipped[id] then
                    if equippedCount < maxSlots then
                        table.insert(toEquip, id)
                        equippedCount = equippedCount + 1
                    end
                end
            end

            local function equipNext(index)
                if index > #toEquip then
                    isEquippingRelic = false
                    return
                end
                RelicsService:EquipRelic(toEquip[index]):andThen(function()
                    equipNext(index + 1)
                end):catch(function()
                    equipNext(index + 1)
                end)
            end

            equipNext(1)
        end):catch(function()
            isEquippingRelic = false
        end)
    end)
end
task.spawn(function()
    while true do
        autoEquipBestRelic()
        task.wait(1)
    end
end)

local function isAuraWorldLocked(name)
    local req = AurasConfig.RequiredWorldByAuraId and AurasConfig.RequiredWorldByAuraId[name]
    if not req then return false end
    local highest = tonumber(LocalPlayer:GetAttribute("HighestVisitedWorld")) or 1
    return highest < math.max(1, tonumber(req))
end
local function isAuraOwned(name)
    if not auraState or not auraState.Items then return false end
    local item = auraState.Items[name]
    return item and item.Owned == true
end
local function isAuraEquipped(name)
    if not auraState or not auraState.Items then return false end
    local item = auraState.Items[name]
    return item and item.Equipped == true
end
local function findCheapestAffordableUnownedAura()
    local bestName, bestPrice = nil, math.huge
    for _, auraName in ipairs(auraNames) do
        if not isAuraOwned(auraName) and not isAuraWorldLocked(auraName) then
            local price = tonumber(AurasConfig.Items[auraName].Price) or 0
            if price <= money and price < bestPrice then
                bestName = auraName
                bestPrice = price
            end
        end
    end
    return bestName
end
local function findBestOwnedAura()
    local bestName, bestBoost = nil, -math.huge
    for auraName, data in pairs(AurasConfig.Items) do
        if isAuraOwned(auraName) then
            local boost = tonumber(data.StrengthMultiplier) or 0
            if boost > bestBoost then
                bestBoost = boost
                bestName = auraName
            end
        end
    end
    return bestName
end
local function buyOrToggleAura(auraName)
    if isBusyAura then return end
    isBusyAura = true
    pcall(function()
        local res = AuraService:BuyOrToggleAura(auraName)
        if type(res) == "table" and res.andThen then
            res:andThen(function(state)
                isBusyAura = false
                if type(state) == "table" and state.State then
                    updateAuraState(state.State)
                else
                    refreshAuraState()
                end
            end):catch(function()
                isBusyAura = false
                refreshAuraState()
            end)
        else
            isBusyAura = false
            refreshAuraState()
        end
    end)
end

end)

local function isCutterWorldUnlocked(name)
    local data = cuttersData[name]
    if not data then return false end
    local requiredWorld = tonumber(data.RequiredWorld) or 1
    local highestWorld = tonumber(LocalPlayer:GetAttribute("HighestVisitedWorld")) or 1
    return highestWorld >= requiredWorld
end
local function findCheapestAffordableUnownedCutter()
    local bestName, bestPrice = nil, math.huge
    for name, data in pairs(cuttersData) do
        if not ownedCutters[name] and isCutterWorldUnlocked(name) then
            local price = tonumber(data.Price) or 0
            if price <= money and price < bestPrice then
                bestName = name
                bestPrice = price
            end
        end
    end
    return bestName
end
local function findBestOwnedCutter()
    local bestName, bestDamage = nil, -math.huge
    for name, data in pairs(cuttersData) do
        if ownedCutters[name] then
            local dmg = tonumber(data.DamageMult) or 0
            if dmg > bestDamage then
                bestDamage = dmg
                bestName = name
            end
        end
    end
    return bestName
end

    end)
end

local function getLootPriority(rarity)
    for i, r in ipairs(rarityOrder) do
        if r == rarity then return i end
    end
    return #rarityOrder + 1
end
local function getLootPosition(loot)
    if loot:IsA("Model") then
        return loot:GetPivot().Position
    elseif loot:IsA("BasePart") then
        return loot.Position
    else
        return loot:GetPivot().Position
    end
end
local function findPrompt(loot)
    local direct = loot:FindFirstChild("PickupPrompt")
    if direct and direct:IsA("ProximityPrompt") then return direct end
    for _, child in ipairs(loot:GetDescendants()) do
        if child.Name == "PickupPrompt" and child:IsA("ProximityPrompt") then return child end
    end
    return nil
end
local function triggerPrompt(prompt)
    if fireproximityprompt then
        fireproximityprompt(prompt)
    elseif prompt.InputHoldBegin then
        prompt:InputHoldBegin()
        task.wait(0.1)
        prompt:InputHoldEnd()
    else
        pcall(function()
            game:GetService("ProximityPromptService"):Prompt(prompt)
        end)
    end
end
local function isLootValid(loot)
    return loot and loot.Parent and CollectionService:HasTag(loot, "SpawnedLoot")
end
local function getBestLoot()
    local lootList = {}
    for _, loot in ipairs(CollectionService:GetTagged("SpawnedLoot")) do
        if loot and loot.Parent then
            local rarity = loot:GetAttribute("LootRarity")
            if rarity and selectedLootRarityMap[rarity] then
                table.insert(lootList, loot)
            end
        end
    end
    if #lootList == 0 then return nil end
    table.sort(lootList, function(a, b)
        local pa = getLootPriority(a:GetAttribute("LootRarity"))
        local pb = getLootPriority(b:GetAttribute("LootRarity"))
        if pa ~= pb then return pa < pb end
        local hrp = getHRP()
        if hrp then
            return (getLootPosition(a) - hrp.Position).Magnitude < (getLootPosition(b) - hrp.Position).Magnitude
        end
        return false
    end)
    return lootList[1]
end
task.spawn(function()
    while true do
        if autoCollectLootEnabled then
            refreshBackpackFull()
            if backpackFull then
                returnToSavedPosition()
                while autoCollectLootEnabled and backpackFull do
                    task.wait(1)
                    refreshBackpackFull()
                end
                if not autoCollectLootEnabled then break end
            end
            local target = getBestLoot()
            if target then
                while target and isLootValid(target) and autoCollectLootEnabled do
                    local better = getBestLoot()
                    if better and better ~= target and getLootPriority(better:GetAttribute("LootRarity")) < getLootPriority(target:GetAttribute("LootRarity")) then
                        target = better
                    end
                    if backpackFull then
                        returnToSavedPosition()
                        while autoCollectLootEnabled and backpackFull do
                            task.wait(1)
                            refreshBackpackFull()
                        end
                        if not autoCollectLootEnabled then break end
                        target = getBestLoot()
                        if not target then break end
                    end
                    local pos = getLootPosition(target)
                    local hrp = getHRP()
                    if not hrp then
                        task.wait(0.2)
                        break
                    end
                    hrp.CFrame = CFrame.new(pos + Vector3.new(0, 2, 0))
                    task.wait(0.2)
                    local prompt = findPrompt(target)
                    if prompt then triggerPrompt(prompt) end
                    task.wait(0.2)
                    if not isLootValid(target) then break end
                end
            else
                task.wait(0.2)
            end
        else
            task.wait(0.2)
        end
    end
end)
