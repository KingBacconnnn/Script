local TWEEN_FAST = TweenInfo.new(0.10, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local TWEEN_NORMAL = TweenInfo.new(0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
local TWEEN_SLOW = TweenInfo.new(0.28, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

local function tween(instance, properties, duration)
    local info = duration and (duration <= 0.12 and TWEEN_FAST or duration >= 0.25 and TWEEN_SLOW or TweenInfo.new(duration, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)) or TWEEN_NORMAL
    local ok, result = pcall(function()
        local t = TweenService:Create(instance, info, properties)
        t:Play()
        return t
    end)
    return ok and result or nil
end

local function safeCall(callback, ...)
    if type(callback) ~= "function" then
        return true
    end
    local ok, err = pcall(callback, ...)
    if not ok then
        warn("[VeloxUI] Callback error:", err)
    end
    return ok
end

local function newInstance(className, props, parent)
    local obj = Instance.new(className)
    for key, value in pairs(props or {}) do
        pcall(function()
            obj[key] = value
        end)
    end
    if parent then
        obj.Parent = parent
    end
    return obj
end

local function radius(instance, amount)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, amount or 9)
    c.Parent = instance
    return c
end

local function outline(instance, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = instance
    return s
end

local function shade(color, amount)
    local h, s, v = color:ToHSV()
    return Color3.fromHSV(h, s, math.clamp(v + amount, 0, 1))
end

local function addPadding(instance, left, top, right, bottom)
    local pad = Instance.new("UIPadding")
    pad.PaddingLeft = UDim.new(0, left or 0)
    pad.PaddingTop = UDim.new(0, top or 0)
    pad.PaddingRight = UDim.new(0, right or left or 0)
    pad.PaddingBottom = UDim.new(0, bottom or top or 0)
    pad.Parent = instance
    return pad
end

local function mergeTheme(base, custom)
    local result = {}
    for k, v in pairs(base or DEFAULTS) do
        result[k] = v
    end
    for k, v in pairs(custom or {}) do
        result[k] = v
    end
    return result
end

local function getViewport()
    local camera = Workspace.CurrentCamera
    return camera and camera.ViewportSize or Vector2.new(1280, 720)
end

local function isCompact()
    local size = getViewport()
    return (UserInputService.TouchEnabled and not UserInputService.MouseEnabled) or size.X < 760
end

local function makeIcon(parent, icon, size, color, z)
    if not icon then
        return nil
    end

    local raw = tostring(icon)
    if raw:match("^rbxassetid://") or raw:match("^%d+$") then
        if raw:match("^%d+$") then
            raw = "rbxassetid://" .. raw
        end
        return newInstance("ImageLabel", {
            BackgroundTransparency = 1,
            Image = raw,
            Size = UDim2.fromOffset(size or 18, size or 18),
            ImageColor3 = color,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromScale(0.5, 0.5),
            ZIndex = z or 5,
        }, parent)
    end

    return newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Text = iconGlyph(icon) or "•",
        Size = UDim2.fromOffset(size or 18, size or 18),
        Font = Enum.Font.GothamBold,
        TextSize = size and math.max(size - 2, 10) or 16,
        TextColor3 = color,
        TextXAlignment = Enum.TextXAlignment.Center,
        TextYAlignment = Enum.TextYAlignment.Center,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0.5, 0.5),
        ZIndex = z or 5,
    }, parent)
end

local Cleaner = {}
Cleaner.__index = Cleaner

function Cleaner.new()
    return setmetatable({ items = {} }, Cleaner)
end

function Cleaner:Add(item)
    table.insert(self.items, item)
    return item
end

function Cleaner:Clean()
    for i = #self.items, 1, -1 do
        local item = self.items[i]
        self.items[i] = nil
        if typeof(item) == "RBXScriptConnection" then
            pcall(function() item:Disconnect() end)
        elseif type(item) == "function" then
            pcall(item)
        elseif typeof(item) == "Instance" then
            pcall(function() item:Destroy() end)
        elseif type(item) == "table" and type(item.Destroy) == "function" then
            pcall(function() item:Destroy() end)
        end
    end
end

local InputBus = {
    Captures = {},
    Toggles = {},
    KeyListener = nil,
    MouseListener = nil,
}

InputBus.KeyListener = UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        for i = #InputBus.Captures, 1, -1 do
            local capture = InputBus.Captures[i]
            if capture and not capture.cancelled then
                table.remove(InputBus.Captures, i)
                safeCall(capture.callback, input.KeyCode.Name, input.KeyCode)
                return
            end
        end
    end

    if gameProcessed then
        return
    end

    local token = input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode.Name
        or input.UserInputType == Enum.UserInputType.MouseButton1 and "MouseLeft"
        or input.UserInputType == Enum.UserInputType.MouseButton2 and "MouseRight"
        or nil

    if not token then
        return
    end

    for _, binding in ipairs(InputBus.Toggles) do
        if binding and binding.key == token then
            safeCall(binding.callback)
        end
    end
end)

function InputBus:RegisterToggle(key, callback)
    local item = { key = key, callback = callback }
    table.insert(self.Toggles, item)
    return function()
        local index = table.find(self.Toggles, item)
        if index then
            table.remove(self.Toggles, index)
        end
    end
end

function InputBus:Capture(callback)
    local item = { callback = callback, cancelled = false }
    table.insert(self.Captures, item)
    return function()
        item.cancelled = true
        local index = table.find(self.Captures, item)
        if index then
            table.remove(self.Captures, index)
        end
    end
end

local DragBus = {
    activeMove = nil,
    activeEnd = nil,
}

UserInputService.InputChanged:Connect(function(input)
    if DragBus.activeMove and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        pcall(DragBus.activeMove, input)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if DragBus.activeEnd and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        local ending = DragBus.activeEnd
        DragBus.activeMove = nil
        DragBus.activeEnd = nil
        pcall(ending, input)
    end
end)

function DragBus:Begin(onMove, onEnd)
    self.activeMove = onMove
    self.activeEnd = onEnd or function() end
end

function DragBus:Cancel()
    self.activeMove = nil
    self.activeEnd = nil
end

local function bindButtonFeel(button, getBase, getHover, getPressed)
    local function setBackground(color)
        if button and button.Parent then
            tween(button, { BackgroundColor3 = color }, 0.10)
        end
    end

    button.MouseEnter:Connect(function()
        setBackground(getHover and getHover() or shade(getBase(), 0.06))
    end)

    button.MouseLeave:Connect(function()
        setBackground(getBase())
    end)

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            setBackground(getPressed and getPressed() or shade(getBase(), -0.07))
        end
    end)

    button.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            setBackground(getBase())
        end
    end)
end

local NotificationState = {
    Holder = nil,
    Gui = nil,
    Count = 0,
}

local function getNotificationHolder(theme)
    if NotificationState.Holder and NotificationState.Holder.Parent then
        return NotificationState.Holder
    end

    local gui = newInstance("ScreenGui", {
        Name = "VeloxUI_Notifications",
        IgnoreGuiInset = true,
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
    })

    local parentOk = pcall(function()
        gui.Parent = game:GetService("CoreGui")
    end)

    if not parentOk then
        gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end

    local holder = newInstance("Frame", {
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 1),
        Position = UDim2.new(1, -16, 1, -16),
        Size = UDim2.new(0, math.min(360, math.max(getViewport().X - 32, 240)), 1, -32),
    }, gui)

    newInstance("UIListLayout", {
        FillDirection = Enum.FillDirection.Vertical,
        VerticalAlignment = Enum.VerticalAlignment.Bottom,
        HorizontalAlignment = Enum.HorizontalAlignment.Right,
        Padding = UDim.new(0, 8),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, holder)

    NotificationState.Gui = gui
    NotificationState.Holder = holder
    NotificationState.Theme = theme
    return holder
end

local function createNotification(config)
    config = config or {}
    local theme = config.Theme or VeloxUI.CurrentTheme
    local holder = getNotificationHolder(theme)
    NotificationState.Count += 1

    local variant = tostring(config.Variant or config.Type or "Info"):lower()
    local accent = ({
        success = theme.Positive,
        warning = theme.Warning,
        error = theme.Negative,
        info = theme.Info,
    })[variant] or theme.Accent

    local frame = newInstance("Frame", {
        BackgroundColor3 = theme.Surface,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        ClipsDescendants = true,
        LayoutOrder = NotificationState.Count,
    }, holder)
    radius(frame, 12)
    local frameStroke = outline(frame, theme.Border, 1)

    local accentBar = newInstance("Frame", {
        BackgroundColor3 = accent,
        Size = UDim2.new(0, 3, 1, -18),
        Position = UDim2.new(0, 0, 0, 9),
    }, frame)
    radius(accentBar, 3)

    local title = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 14, 0, 10),
        Size = UDim2.new(1, -42, 0, 20),
        Font = Enum.Font.GothamBold,
        Text = config.Title or "Notification",
        TextColor3 = theme.Text,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, frame)

    local content = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 14, 0, 31),
        Size = UDim2.new(1, -28, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        Font = Enum.Font.Gotham,
        Text = config.Content or config.Text or "",
        TextColor3 = theme.SubText,
        TextSize = 13,
        TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, frame)

    local close = newInstance("TextButton", {
        BackgroundColor3 = theme.SurfaceHover,
        BackgroundTransparency = 0.35,
        Position = UDim2.new(1, -32, 0, 8),
        Size = UDim2.fromOffset(24, 24),
        Text = "×",
        Font = Enum.Font.GothamBold,
        TextSize = 16,
        TextColor3 = theme.SubText,
        AutoButtonColor = false,
    }, frame)
    radius(close, 7)

    local closed = false
    local function destroy()
        if closed then
            return
        end
        closed = true

        local t = tween(frame, { BackgroundTransparency = 1 }, 0.15)
        tween(title, { TextTransparency = 1 }, 0.15)
        tween(content, { TextTransparency = 1 }, 0.15)
        tween(accentBar, { BackgroundTransparency = 1 }, 0.15)
        if t then
            t.Completed:Connect(function()
                if frame and frame.Parent then
                    frame:Destroy()
                end
            end)
        else
            frame:Destroy()
        end
    end

    close.Activated:Connect(destroy)
    bindButtonFeel(close, function() return theme.SurfaceHover end)

    tween(frame, { BackgroundTransparency = 0 }, 0.18)
    tween(frameStroke, { Transparency = 0 }, 0.18)

    local duration = tonumber(config.Duration) or 3
    task.delay(duration, destroy)

    return {
        Instance = frame,
        Close = destroy,
        Destroy = destroy,
        SetTitle = function(_, value)
            title.Text = tostring(value)
        end,
        SetContent = function(_, value)
            content.Text = tostring(value)
        end,
    }
end

function VeloxUI:Notify(config)
    return createNotification(config)
end

local function defaultWindowSize()
    local vp = getViewport()
    if isCompact() then
        return UDim2.fromOffset(
            math.floor(math.clamp(vp.X - 20, 300, 540)),
            math.floor(math.clamp(vp.Y - 44, 260, 720))
        )
    end

    return UDim2.fromOffset(
        math.floor(math.clamp(vp.X * 0.52, 620, 860)),
        math.floor(math.clamp(vp.Y * 0.62, 420, 620))
    )
end

local function centerPosition(size)
    local vp = getViewport()
    return UDim2.fromOffset(
        math.max(math.floor((vp.X - size.X.Offset) / 2), 0),
        math.max(math.floor((vp.Y - size.Y.Offset) / 2), 0)
    )
end

local function clampPosition(window)
    local vp = getViewport()
    local size = window._main.AbsoluteSize
    local pos = window._main.Position
    window._main.Position = UDim2.fromOffset(
        math.clamp(pos.X.Offset, 0, math.max(vp.X - size.X, 0)),
        math.clamp(pos.Y.Offset, 0, math.max(vp.Y - size.Y, 0))
    )
end

local function windowThemeBind(window, instance, property, key)
    local bind = { instance = instance, property = property, key = key }
    table.insert(window._themeBinds, bind)
    if window._theme[key] ~= nil then
        pcall(function()
            instance[property] = window._theme[key]
        end)
    end
    return bind
end

local function setTextTransparency(root, value)
    for _, obj in ipairs(root:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            obj.TextTransparency = value
        elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
            obj.ImageTransparency = value
        end
    end
end

function VeloxUI:CreateWindowBase(config)
    config = config or {}
    local compact = isCompact()
    local metrics = getMetrics(compact)
    local theme = config.Theme

    if type(theme) == "string" then
        theme = VeloxUI.Themes[theme]
    elseif type(theme) ~= "table" then
        theme = VeloxUI.CurrentTheme
    end

    theme = mergeTheme(VeloxUI.CurrentTheme, theme)

    if config.Accent then
        theme.Accent = config.Accent
    end

    if config.AccentHover then
        theme.AccentHover = config.AccentHover
    end

    local size = config.Size or defaultWindowSize()
    local sidebarWidth = config.SidebarWidth or metrics.Sidebar
    local cleaner = Cleaner.new()

    local gui = newInstance("ScreenGui", {
        Name = config.Folder or "VeloxUI",
        ResetOnSpawn = false,
        IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        DisplayOrder = tonumber(config.DisplayOrder) or 20,

    local parentOk = pcall(function()
        gui.Parent = game:GetService("CoreGui")
    end)

    if not parentOk then
        gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end

    local main = newInstance("Frame", {
        Size = size,
        Position = centerPosition(size),
        BackgroundColor3 = theme.Background,
        ClipsDescendants = true,
        Active = true,

        _theme = theme,
        _themeBinds = {},
        _gui = gui,
        _main = main,
        _cleaner = cleaner,
        _tabs = {},
        _activeTab = nil,
        _compact = compact,
        _metrics = metrics,
        _sidebarWidth = sidebarWidth,
        _size = size,
        _minimized = false,
        _fullscreen = false,
        _destroyed = false,
        _visible = true,
        _resizable = config.Resizable ~= false and not compact,
        _closable = config.Closable ~= false,
        _minimizable = config.Minimizable ~= false,
        _searchEnabled = config.Search ~= false,
        _onOpen = {},
        _onClose = {},
        _onDestroy = {},
    }, WindowMeta)

    windowThemeBind(window, main, "BackgroundColor3", "Background")
    windowThemeBind(window, mainStroke, "Color", "Border")

    local scale = newInstance("UIScale", { Scale = 0.96 }, main)
    window._scale = scale

    local overlay = newInstance("Frame", {
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        ZIndex = 70,
    }, gui)

    local header = newInstance("Frame", {
        BackgroundColor3 = theme.Sidebar,
        Size = UDim2.new(1, 0, 0, metrics.TopBar),
        BorderSizePixel = 0,
        Active = true,
    }, main)
    windowThemeBind(window, header, "BackgroundColor3", "Sidebar")

    local headerGlow = newInstance("Frame", {
        BackgroundColor3 = theme.Accent,
        BackgroundTransparency = 0.82,
        Size = UDim2.new(0, 3, 1, -12),
        Position = UDim2.new(0, 0, 0, 6),
    }, header)
    radius(headerGlow, 2)

    local hamburger = newInstance("TextButton", {
        BackgroundColor3 = theme.Surface,
        Size = UDim2.fromOffset(32, 32),
        Position = UDim2.new(0, 8, 0.5, -16),
        Text = "☰",
        Font = Enum.Font.GothamBold,
        TextSize = 17,
        TextColor3 = theme.Text,
        AutoButtonColor = false,
        Visible = compact,
    }, header)
    radius(hamburger, 9)
    windowThemeBind(window, hamburger, "BackgroundColor3", "Surface")
    windowThemeBind(window, hamburger, "TextColor3", "Text")

    local brand = newInstance("Frame", {
        BackgroundTransparency = 1,
        Position = compact and UDim2.new(0, 48, 0, 0) or UDim2.new(0, 14, 0, 0),
        Size = UDim2.new(1, -160, 1, 0),
    }, header)

    local brandIcon = newInstance("Frame", {
        BackgroundColor3 = theme.AccentSoft,
        Size = UDim2.fromOffset(26, 26),
        Position = UDim2.new(0, 0, 0.5, -13),
    }, brand)
    radius(brandIcon, 8)
    windowThemeBind(window, brandIcon, "BackgroundColor3", "AccentSoft")

    makeIcon(brandIcon, iconConfig, 16, theme.Accent)

    local title = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 34, 0, 5),
        Size = UDim2.new(1, -34, 0, 19),
        Font = Enum.Font.GothamBold,
        Text = config.Title or "VeloxUI",
        TextColor3 = theme.Text,
        TextSize = 15,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, brand)
    windowThemeBind(window, title, "TextColor3", "Text")

        local search = newInstance("TextBox", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 30, 0, 0),
            Size = UDim2.new(1, -38, 1, 0),
            PlaceholderText = "Search tabs...",

            local nextCompact = isCompact()

    window._dragHandle = header
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local startMouse = input.Position
            local startPos = main.Position
            DragBus:Begin(function(moveInput)
                if window._destroyed or window._fullscreen then
                    return
                end

                local sizeNow = main.AbsoluteSize

            BackgroundTransparency = 1,

            Position = UDim2.new(1, -18, 1, -18),
            Text = "",
            AutoButtonColor = false,
            ZIndex = 25,
        }, main)

        resizeHandle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                local startInput = input.Position
                local startSize = main.AbsoluteSize
                DragBus:Begin(function(moveInput)
                    local delta = moveInput.Position - startInput
                    local vp = getViewport()
                    local minW = window._compact and 300 or 620
                    local minH = window._compact and 260 or 420
                    local maxW = vp.X - main.AbsolutePosition.X
                    local maxH = vp.Y - main.AbsolutePosition.Y
                    local newW = math.clamp(startSize.X + delta.X, minW, math.max(minW, maxW))
                    local newH = math.clamp(startSize.Y + delta.Y, minH, math.max(minH, maxH))
                    main.Size = UDim2.fromOffset(newW, newH)
                    window._size = main.Size
                end)
            end
        end)
    end

    window._resizeHandle = resizeHandle

    windowThemeBind(window, title, "TextColor3", "Text")
    windowThemeBind(window, headerGlow, "BackgroundColor3", "Accent")

    applyLayout(compact)

    window:SetToggleKey(config.ToggleKey)

    window._cleaner:Add(function()
        DragBus:Cancel()
    end)

    tween(scale, { Scale = 1 }, 0.24)

    return window
end

function WindowMeta:CreateTab(config)
    if self._destroyed then
        return nil
    end

    if type(config) == "string" then
        config = { Title = config }
    end

    config = config or {}

    local theme = self._theme
    local m = self._metrics
    local title = tostring(config.Title or "Tab")

    local button = newInstance("TextButton", {
        BackgroundColor3 = theme.Surface,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, m.TabHeight),
        Text = "",
        AutoButtonColor = false,
    }, self._tabList)
    radius(button, 10)

    local selectedGlow = newInstance("Frame", {
        BackgroundColor3 = theme.Accent,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0.5, -11),
        Size = UDim2.fromOffset(3, 22),
    }, button)
    radius(selectedGlow, 2)

    local iconHolder = newInstance("Frame", {
        BackgroundTransparency = 1,
        Size = UDim2.fromOffset(22, 22),
        Position = UDim2.new(0, 10, 0.5, -11),
    }, button)

    local iconNode = makeIcon(iconHolder, config.Icon, 16, theme.SubText, 3)

    local label = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, config.Icon and 38 or 14, 0, 0),
        Size = UDim2.new(1, -(config.Icon and 48 or 24), 1, 0),
        Font = Enum.Font.GothamMedium,
        Text = title,
        TextColor3 = theme.SubText,
        TextSize = m.TitleSize,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, button)

    local badge
    if config.Badge then
        badge = newInstance("TextLabel", {
            BackgroundColor3 = theme.AccentSoft,
            Size = UDim2.fromOffset(24, 20),
            Position = UDim2.new(1, -30, 0.5, -10),
            Text = tostring(config.Badge),
            Font = Enum.Font.GothamBold,
            TextSize = 10,
            TextColor3 = theme.Accent,
        }, button)
        radius(badge, 7)
    end

    local page = newInstance("ScrollingFrame", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 3,
        BorderSizePixel = 0,
        Visible = false,
    }, self._tabHost)

    addPadding(page, m.Pad, m.Pad + 2, m.Pad, m.Pad)

    local layout = newInstance("UIListLayout", {
        Padding = UDim.new(0, m.Gap),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, page)

    local tab = setmetatable({
        _window = self,
        _button = button,
        _page = page,
        _title = title,
        _config = config,
        _theme = theme,
        _metrics = m,
        _selected = false,
        _cleaner = Cleaner.new(),
        _icon = iconNode,
        _iconHolder = iconHolder,
        _label = label,
        _indicator = selectedGlow,
        _badge = badge,
    }, TabMeta)

    tab._cleaner:Add(button.Activated:Connect(function()
        tab:Select()
    end))

    bindButtonFeel(button, function()
        return tab._selected and theme.SurfaceHover or theme.Surface
    end, function()
        return theme.SurfaceHover
    end)

    table.insert(self._tabs, tab)

    if #self._tabs == 1 then
        tab:Select()
    end

    return tab
end

WindowMeta.Tab = WindowMeta.CreateTab

function TabMeta:Select()
    if not self or not self._window or self._window._destroyed then
        return
    end

    for _, other in ipairs(self._window._tabs) do
        other._selected = false
        other._page.Visible = false
        tween(other._button, {
            BackgroundTransparency = 1,
        }, 0.12)
        tween(other._indicator, {
            BackgroundTransparency = 1,
        }, 0.12)
        tween(other._label, {
            TextColor3 = self._theme.SubText,
        }, 0.12)
        if other._icon then
            if other._icon:IsA("TextLabel") then
                tween(other._icon, { TextColor3 = self._theme.SubText }, 0.12)
            elseif other._icon:IsA("ImageLabel") then
                tween(other._icon, { ImageColor3 = self._theme.SubText }, 0.12)
            end
        end
    end

    self._selected = true
    self._page.Visible = true
    self._window._activeTab = self

    tween(self._button, {
        BackgroundTransparency = 0,
        BackgroundColor3 = self._theme.SurfaceHover,
    }, 0.14)

    tween(self._indicator, {
        BackgroundTransparency = 0,
    }, 0.14)

    tween(self._label, {
        TextColor3 = self._theme.Text,
    }, 0.14)

    if self._icon then
        if self._icon:IsA("TextLabel") then
            tween(self._icon, { TextColor3 = self._theme.Accent }, 0.14)
        elseif self._icon:IsA("ImageLabel") then
            tween(self._icon, { ImageColor3 = self._theme.Accent }, 0.14)
        end
    end

    if self._window._compact and self._window._setDrawer then
        self._window._setDrawer(false)
    end
end

function TabMeta:SetTitle(title)
    self._title = tostring(title)
    self._label.Text = self._title
    return self
end

function TabMeta:SetIcon(icon)
    if self._icon then
        self._icon:Destroy()
        self._icon = nil
    end
    if self._iconHolder then
        self._icon = makeIcon(self._iconHolder, icon, 16, self._theme.SubText, 3)
    end
    return self
end

function TabMeta:SetBadge(value)
    if self._badge then
        self._badge:Destroy()
        self._badge = nil
    end
    if value ~= nil then
        local badge = newInstance("TextLabel", {
            BackgroundColor3 = self._theme.AccentSoft,
            Size = UDim2.fromOffset(24, 20),
            Position = UDim2.new(1, -30, 0.5, -10),
            Text = tostring(value),
            Font = Enum.Font.GothamBold,
            TextSize = 10,
            TextColor3 = self._theme.Accent,
        }, self._button)
        radius(badge, 7)
        self._badge = badge
    end
    return self
end

local function createRow(tab, title, desc, height)
    local m = tab._metrics
    local theme = tab._theme

    local row = newInstance("Frame", {
        BackgroundColor3 = theme.Surface,
        Size = UDim2.new(1, 0, 0, height or m.Row),
    }, tab._page)
    radius(row, m.Radius)

    local accent = newInstance("Frame", {
        BackgroundColor3 = theme.Accent,
        BackgroundTransparency = 1,
        Size = UDim2.new(0, 2, 0, math.max((height or m.Row) - 20, 10)),
        Position = UDim2.new(0, 0, 0, 10),
    }, row)
    radius(accent, 2)

    local textZone = newInstance("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 14, 0, 0),
        Size = UDim2.new(1 - m.ControlWidth, -18, 1, 0),
    }, row)

    local titleLabel = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = desc and UDim2.new(0, 0, 0, 8) or UDim2.new(),
        Size = desc and UDim2.new(1, 0, 0, 18) or UDim2.new(1, 0, 1, 0),
        Font = Enum.Font.GothamMedium,
        Text = tostring(title or ""),
        TextColor3 = theme.Text,
        TextSize = m.TitleSize,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, textZone)

    local descLabel
    if desc then
        descLabel = newInstance("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 0, 0, 27),
            Size = UDim2.new(1, 0, 0, 16),
            Font = Enum.Font.Gotham,
            Text = tostring(desc),
            TextColor3 = theme.SubText,
            TextSize = m.DescSize,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextTruncate = Enum.TextTruncate.AtEnd,
        }, textZone)
    end

    local control = newInstance("Frame", {
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -12, 0.5, 0),
        Size = UDim2.new(m.ControlWidth, 0, 0, math.min(32, (height or m.Row) - 16)),
    }, row)

    return row, control, titleLabel, descLabel
end

function TabMeta:CreateButton(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme
    local variant = tostring(config.Variant or "Solid"):lower()

    local baseColor = ({
        solid = theme.Accent,
        primary = theme.Accent,
        ghost = theme.SurfaceHover,
        secondary = theme.SurfaceHover,
        danger = theme.Negative,
        success = theme.Positive,
        warning = theme.Warning,
    })[variant] or theme.Accent

    local textColor = Color3.new(1, 1, 1)
    if variant == "ghost" or variant == "secondary" then
        textColor = theme.Text
    elseif variant == "warning" then
        textColor = Color3.new(0.12, 0.12, 0.12)
    end

    local button = newInstance("TextButton", {
        BackgroundColor3 = baseColor,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false,
    }, control)
    radius(button, self._metrics.Radius - 2)

    if variant == "ghost" or variant == "secondary" then
        outline(button, theme.Border, 1)
    end

    if config.Icon then
        local iconFrame = newInstance("Frame", {
            BackgroundTransparency = 1,
            Size = UDim2.fromOffset(18, 18),
            Position = UDim2.new(0, 10, 0.5, -9),
        }, button)
        makeIcon(iconFrame, config.Icon, 15, config.IconColor or textColor)
    end

    local buttonText = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, config.Icon and 30 or 8, 0, 0),
        Size = UDim2.new(1, config.Icon and -38 or -16, 1, 0),
        Font = Enum.Font.GothamBold,
        Text = config.ButtonText or config.Text or "Run",
        TextColor3 = textColor,
        TextSize = self._metrics.DescSize + 1,
        TextXAlignment = Enum.TextXAlignment.Center,
    }, button)

    local locked = config.Locked == true
    local function setLocked(value)
        locked = value == true
        button.Active = not locked
        button.AutoButtonColor = false
        tween(button, {
            BackgroundTransparency = locked and 0.45 or 0,
        }, 0.15)
    end

    bindButtonFeel(button, function() return baseColor end)
    button.Activated:Connect(function()
        if locked then
            return
        end
        safeCall(config.Callback)
    end)

    return {
        Instance = row,
        Button = button,
        Lock = function() setLocked(true) end,
        Unlock = function() setLocked(false) end,
        Set = function(_, value)
            buttonText.Text = tostring(value)
        end,
        Get = function()
            return buttonText.Text
        end,
        Destroy = function()
            row:Destroy()
        end,
    }
end

function TabMeta:CreateToggle(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme
    local state = config.Value == true or config.Default == true
    local switchSize = self._metrics.Switch

    local switch = newInstance("Frame", {
        BackgroundColor3 = state and theme.Accent or theme.BorderSoft,
        Size = UDim2.fromOffset(switchSize.X, switchSize.Y),
        Position = UDim2.new(1, -switchSize.X, 0.5, -switchSize.Y / 2),
    }, control)
    radius(switch, switchSize.Y / 2)

    local knobSize = switchSize.Y - 6
    local knob = newInstance("Frame", {
        BackgroundColor3 = Color3.new(1, 1, 1),
        Size = UDim2.fromOffset(knobSize, knobSize),
        Position = state and UDim2.new(1, -knobSize - 3, 0.5, -knobSize / 2) or UDim2.new(0, 3, 0.5, -knobSize / 2),
    }, switch)
    radius(knob, knobSize / 2)

    local hit = newInstance("TextButton", {
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Text = "",
        AutoButtonColor = false,
    }, switch)

    local locked = config.Locked == true

    local function setState(value, fire)
        state = value == true
        tween(switch, {
            BackgroundColor3 = state and theme.Accent or theme.BorderSoft,
        }, 0.14)
        tween(knob, {
            Position = state and UDim2.new(1, -knobSize - 3, 0.5, -knobSize / 2) or UDim2.new(0, 3, 0.5, -knobSize / 2),
        }, 0.14)
        if fire and not locked then
            safeCall(config.Callback, state)
        end
    end

    hit.Activated:Connect(function()
        if not locked then
            setState(not state, true)
        end
    end)

    return {
        Instance = row,
        Set = function(_, value)
            setState(value, false)
        end,
        Get = function()
            return state
        end,
        Lock = function()
            locked = true
            tween(switch, { BackgroundTransparency = 0.35 }, 0.15)
        end,
        Unlock = function()
            locked = false
            tween(switch, { BackgroundTransparency = 0 }, 0.15)
        end,
        Destroy = function()
            row:Destroy()
        end,
    }
end

function TabMeta:CreateCheckbox(config)
    return self:CreateToggle(config)
end

function TabMeta:CreateSlider(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height or self._metrics.SliderRow)
    local theme = self._theme
    local min = tonumber(config.Min) or 0
    local max = tonumber(config.Max) or 100
    local step = tonumber(config.Step) or 1
    local value = tonumber(config.Value or config.Default) or min

    if max < min then
        min, max = max, min
    end

    if step <= 0 then
        step = 1
    end

    local label = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, -2),
        Size = UDim2.fromOffset(80, 16),
        Font = Enum.Font.GothamBold,
        Text = tostring(value),
        TextColor3 = theme.Accent,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Right,
    }, control)

    local hit = newInstance("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 20),
        Size = UDim2.new(1, 0, 0, 24),
    }, control)

    local track = newInstance("Frame", {
        BackgroundColor3 = theme.BorderSoft,
        Position = UDim2.new(0, 0, 0.5, -3),
        Size = UDim2.new(1, 0, 0, 6),
    }, hit)
    radius(track, 3)

    local fill = newInstance("Frame", {
        BackgroundColor3 = theme.Accent,
        Size = UDim2.new(0, 0, 1, 0),
    }, track)
    radius(fill, 3)

    local thumb = newInstance("Frame", {
        BackgroundColor3 = Color3.new(1, 1, 1),
        Size = UDim2.fromOffset(14, 14),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
    }, track)
    radius(thumb, 7)

    local function apply(raw, fire)
        raw = math.clamp(raw, min, max)
        raw = math.floor((raw - min) / step + 0.5) * step + min
        raw = math.clamp(raw, min, max)

        value = raw
        local ratio = (raw - min) / math.max(max - min, 1e-6)

        tween(fill, { Size = UDim2.new(ratio, 0, 1, 0) }, 0.06)
        tween(thumb, { Position = UDim2.new(ratio, 0, 0.5, 0) }, 0.06)
        label.Text = tostring(config.Decimals and tonumber(string.format("%." .. tostring(config.Decimals) .. "f", raw)) or raw)

        if fire then
            safeCall(config.Callback, value)
        end
    end

    local function ratioFromInput(input)
        return math.clamp(
            (input.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1),
            0,
            1
        )
    end

    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            apply(min + ratioFromInput(input) * (max - min), true)
            DragBus:Begin(function(moveInput)
                apply(min + ratioFromInput(moveInput) * (max - min), true)
            end)
        end
    end)

    apply(value, false)

    return {
        Instance = row,
        Set = function(_, v)
            apply(tonumber(v) or min, true)
        end,
        Get = function()
            return value
        end,
        Destroy = function()
            row:Destroy()
        end,
    }
end

function TabMeta:CreateDropdown(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme
    local values = table.clone(config.Values or config.Options or {})
    local multi = config.Multi == true
    local selected = {}

    local defaultValue = config.Value
    if defaultValue == nil then
        defaultValue = config.Default
    end

    if multi then
        if type(defaultValue) == "table" then
            for _, v in ipairs(defaultValue) do
                selected[tostring(v)] = v
            end
        elseif defaultValue ~= nil then
            selected[tostring(defaultValue)] = defaultValue
        end
    end

    local singleValue = multi and nil or defaultValue
    if not multi and singleValue == nil then
        singleValue = values[1]
    end

    local field = newInstance("TextButton", {
        BackgroundColor3 = theme.Input,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false,
    }, control)
    radius(field, self._metrics.Radius - 2)
    outline(field, theme.Border, 1)

    local label = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 10, 0, 0),
        Size = UDim2.new(1, -30, 1, 0),
        Font = Enum.Font.Gotham,
        Text = tostring(singleValue or "Select..."),
        TextColor3 = theme.Text,
        TextSize = self._metrics.DescSize + 1,
        TextTruncate = Enum.TextTruncate.AtEnd,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, field)

    local caret = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -24, 0, 0),
        Size = UDim2.fromOffset(20, 32),
        Text = "⌄",
        Font = Enum.Font.GothamBold,
        TextColor3 = theme.Muted,
        TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Center,
        TextYAlignment = Enum.TextYAlignment.Center,
    }, field)

    local popup = newInstance("Frame", {
        BackgroundColor3 = theme.Surface,
        Visible = false,
        ZIndex = 80,
        ClipsDescendants = true,
    }, self._window._overlay or self._window._gui)
    radius(popup, 10)
    outline(popup, theme.Border, 1)

    local list = newInstance("ScrollingFrame", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 2,
        BorderSizePixel = 0,
        ZIndex = 81,
    }, popup)
    addPadding(list, 6, 6, 6, 6)

    newInstance("UIListLayout", {
        Padding = UDim.new(0, 4),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, list)

    local closeCatcher = newInstance("TextButton", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false,
        Visible = false,
        ZIndex = 79,
    }, self._window._overlay or self._window._gui)

    local function selectedText()
        if not multi then
            return tostring(singleValue or "Select...")
        end

        local names = {}
        for _, v in pairs(selected) do
            table.insert(names, tostring(v))
        end
        table.sort(names)
        return #names > 0 and table.concat(names, ", ") or "Select..."
    end

    local function setValue(value, fire)
        if multi then
            local key = tostring(value)
            if selected[key] ~= nil then
                selected[key] = nil
            else
                selected[key] = value
            end
        else
            singleValue = value
        end

        label.Text = selectedText()
        if fire then
            safeCall(config.Callback, multi and selected or singleValue)
        end
    end

    local function rebuild()
        for _, child in ipairs(list:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end

        for _, value in ipairs(values) do
            local button = newInstance("TextButton", {
                BackgroundColor3 = theme.SurfaceHover,
                BackgroundTransparency = 1,
                Size = UDim2.new(1, 0, 0, self._window._compact and 36 or 32),
                Text = tostring(value),
                Font = Enum.Font.Gotham,
                TextSize = self._metrics.DescSize + 1,
                TextColor3 = theme.Text,
                TextXAlignment = Enum.TextXAlignment.Left,
                AutoButtonColor = false,
                ZIndex = 82,
            }, list)
            radius(button, 7)
            addPadding(button, 10, 0, 10, 0)

            button.Activated:Connect(function()
                setValue(value, true)
                if not multi then
                    popup.Visible = false
                    closeCatcher.Visible = false
                else
                    tween(button, {
                        BackgroundTransparency = selected[tostring(value)] ~= nil and 0 or 1,
                    }, 0.10)
                end
            end)

            button.MouseEnter:Connect(function()
                tween(button, { BackgroundTransparency = 0.85 }, 0.10)
            end)

            button.MouseLeave:Connect(function()
                tween(button, {
                    BackgroundTransparency = selected[tostring(value)] ~= nil and 0.72 or 1,
                }, 0.10)
            end)
        end
    end

    local function positionPopup()
        local vp = getViewport()
        local anchor = field.AbsolutePosition
        local anchorSize = field.AbsoluteSize
        local width = math.max(field.AbsoluteSize.X, 180)
        local height = math.min(#values, config.MaxVisible or 6) * (self._window._compact and 36 or 32) + 12
        width = math.min(width, vp.X - 20)
        height = math.min(height, vp.Y - 20)

        local x = math.clamp(anchor.X, 10, math.max(vp.X - width - 10, 10))
        local y = anchor.Y + anchorSize.Y + 5

        if y + height > vp.Y - 10 then
            y = anchor.Y - height - 5
        end

        y = math.clamp(y, 10, math.max(vp.Y - height - 10, 10))

        popup.Size = UDim2.fromOffset(width, height)
        popup.Position = UDim2.fromOffset(x, y)
    end

    rebuild()

    local open = false
    local function setOpen(value)
        open = value
        if open then
            rebuild()
            positionPopup()
            popup.Visible = true
            closeCatcher.Visible = true
            tween(popup, { BackgroundTransparency = 0 }, 0.12)
        else
            popup.Visible = false
            closeCatcher.Visible = false
        end
    end

    field.Activated:Connect(function()
        setOpen(not open)
    end)

    closeCatcher.Activated:Connect(function()
        setOpen(false)
    end)

    local object = {
        Instance = row,
        Set = function(_, value)
            if multi then
                table.clear(selected)
                if type(value) == "table" then
                    for _, v in ipairs(value) do
                        selected[tostring(v)] = v
                    end
                elseif value ~= nil then
                    selected[tostring(value)] = value
                end
                label.Text = selectedText()
            else
                singleValue = value
                label.Text = tostring(value)
            end
        end,
        Get = function()
            if multi then
                local result = {}
                for _, v in pairs(selected) do
                    table.insert(result, v)
                end
                return result
            end
            return singleValue
        end,
        Refresh = function(_, newValues)
            values = table.clone(newValues or {})
            if popup.Visible then
                positionPopup()
            end
            rebuild()
        end,
        Select = function(_, value)
            if multi then
                selected[tostring(value)] = value
            else
                singleValue = value
            end
            label.Text = selectedText()
            return object
        end,
        Open = function()
            setOpen(true)
        end,
        Close = function()
            setOpen(false)
        end,
        Lock = function()
            field.Active = false
            field.BackgroundTransparency = 0.4
        end,
        Unlock = function()
            field.Active = true
            field.BackgroundTransparency = 0
        end,
        Destroy = function()
            popup:Destroy()
            closeCatcher:Destroy()
            row:Destroy()
        end,
    }

    return object
end

function TabMeta:CreateInput(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme

    local box = newInstance("TextBox", {
        BackgroundColor3 = theme.Input,
        Size = UDim2.new(1, 0, 1, 0),
        Text = tostring(config.Value or ""),
        PlaceholderText = tostring(config.Placeholder or "Enter text..."),
        Font = Enum.Font.Gotham,
        TextColor3 = theme.Text,
        PlaceholderColor3 = theme.Muted,
        TextSize = self._metrics.DescSize + 1,
        ClearTextOnFocus = config.ClearTextOnFocus == true,
        TextXAlignment = Enum.TextXAlignment.Left,
        ClipsDescendants = true,
    }, control)
    radius(box, self._metrics.Radius - 2)
    local boxStroke = outline(box, theme.Border, 1)
    addPadding(box, 10, 0, 10, 0)

    box.Focused:Connect(function()
        tween(boxStroke, { Color = theme.Accent }, 0.12)
    end)

    box.FocusLost:Connect(function()
        tween(boxStroke, { Color = theme.Border }, 0.12)
        safeCall(config.Callback, box.Text)
    end)

    if type(config.OnChanged) == "function" then
        box:GetPropertyChangedSignal("Text"):Connect(function()
            safeCall(config.OnChanged, box.Text)
        end)
    end

    if config.Numeric then
        box:GetPropertyChangedSignal("Text"):Connect(function()
            local filtered = box.Text:gsub("[^%d%.%-]", "")
            if filtered ~= box.Text then
                box.Text = filtered
            end
        end)
    end

    return {
        Instance = row,
        Set = function(_, value)
            box.Text = tostring(value)
        end,
        Get = function()
            return box.Text
        end,
        Focus = function()
            box:CaptureFocus()
        end,
        Clear = function()
            box.Text = ""
        end,
        Destroy = function()
            row:Destroy()
        end,
    }
end

function TabMeta:CreateKeybind(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme
    local current = config.Value or "None"
    local listening = false
    local cancelCapture
    local locked = config.Locked == true

    local button = newInstance("TextButton", {
        BackgroundColor3 = theme.Input,
        Size = UDim2.new(1, 0, 1, 0),
        Text = tostring(current),
        Font = Enum.Font.GothamBold,
        TextColor3 = theme.Text,
        TextSize = self._metrics.DescSize + 1,
        AutoButtonColor = false,
    }, control)
    radius(button, self._metrics.Radius - 2)
    outline(button, theme.Border, 1)

    local function setKey(value, fire)
        current = tostring(value or "None")
        button.Text = listening and "..." or current
        if fire then
            safeCall(config.Callback, current)
        end
    end

    button.Activated:Connect(function()
        if locked or listening then
            return
        end

        listening = true
        button.Text = "Press a key..."

        cancelCapture = InputBus:Capture(function(name)
            listening = false
            setKey(name, true)
            cancelCapture = nil
        end)
    end)

    local destroy = function()
        if cancelCapture then
            cancelCapture()
            cancelCapture = nil
        end
        row:Destroy()
    end

    return {
        Instance = row,
        Set = function(_, value)
            setKey(value, false)
        end,
        Get = function()
            return current
        end,
        Lock = function()
            locked = true
        end,
        Unlock = function()
            locked = false
        end,
        Destroy = destroy,
    }
end

function TabMeta:CreateColorpicker(config)
    config = config or {}
    local row, control = createRow(self, config.Title, config.Desc, config.Height)
    local theme = self._theme
    local color = config.Default or config.Value or Color3.fromRGB(255, 255, 255)
    local rgb = { math.floor(color.R * 255 + 0.5), math.floor(color.G * 255 + 0.5), math.floor(color.B * 255 + 0.5) }

    local preview = newInstance("TextButton", {
        BackgroundColor3 = color,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false,
    }, control)
    radius(preview, self._metrics.Radius - 2)
    outline(preview, theme.Border, 1)

    local swatch = newInstance("Frame", {
        BackgroundColor3 = shade(color, -0.08),
        Size = UDim2.fromOffset(30, 18),
        Position = UDim2.new(1, -36, 0.5, -9),
    }, preview)
    radius(swatch, 6)

    local popup = newInstance("Frame", {
        BackgroundColor3 = theme.Surface,
        Size = UDim2.fromOffset(230, 152),
        Visible = false,
        ZIndex = 85,
    }, self._window._overlay)
    radius(popup, 11)
    outline(popup, theme.Border, 1)
    addPadding(popup, 10, 10, 10, 10)

    local popupTitle = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, 18),
        Font = Enum.Font.GothamBold,
        Text = "Color",
        TextColor3 = theme.Text,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 86,
    }, popup)

    local hex = newInstance("TextBox", {
        BackgroundColor3 = theme.Input,
        Position = UDim2.new(0, 0, 0, 25),
        Size = UDim2.new(1, 0, 0, 28),
        Text = "",
        PlaceholderText = "#RRGGBB",
        Font = Enum.Font.Gotham,
        TextColor3 = theme.Text,
        PlaceholderColor3 = theme.Muted,
        TextSize = 11,
        ZIndex = 86,
        ClearTextOnFocus = false,
    }, popup)
    radius(hex, 8)
    outline(hex, theme.BorderSoft, 1)
    addPadding(hex, 8, 0, 8, 0)

    local function toHex()
        return string.format("#%02X%02X%02X", rgb[1], rgb[2], rgb[3])
    end

    local function applyColor(fire)
        color = Color3.fromRGB(
            math.clamp(math.floor(rgb[1] + 0.5), 0, 255),
            math.clamp(math.floor(rgb[2] + 0.5), 0, 255),
            math.clamp(math.floor(rgb[3] + 0.5), 0, 255)
        )
        preview.BackgroundColor3 = color
        swatch.BackgroundColor3 = shade(color, -0.08)
        hex.Text = toHex()

        if fire then
            safeCall(config.Callback, color)
        end
    end

    local function hexToRgb(text)
        text = text:gsub("#", "")
        if #text ~= 6 then
            return false
        end
        local r = tonumber(text:sub(1, 2), 16)
        local g = tonumber(text:sub(3, 4), 16)
        local b = tonumber(text:sub(5, 6), 16)
        if r and g and b then
            rgb = { r, g, b }
            applyColor(true)
            return true
        end
        return false
    end

    local function channel(name, index, y)
        local rowFrame = newInstance("Frame", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 0, 0, y),
            Size = UDim2.new(1, 0, 0, 24),
            ZIndex = 86,
        }, popup)

        newInstance("TextLabel", {
            BackgroundTransparency = 1,
            Size = UDim2.fromOffset(16, 24),
            Text = name,
            Font = Enum.Font.GothamBold,
            TextColor3 = theme.Text,
            TextSize = 11,
            ZIndex = 87,
        }, rowFrame)

        local track = newInstance("Frame", {
            BackgroundColor3 = theme.BorderSoft,
            Position = UDim2.new(0, 20, 0.5, -3),
            Size = UDim2.new(1, -20, 0, 6),
            ZIndex = 87,
        }, rowFrame)
        radius(track, 3)

        local fill = newInstance("Frame", {
            BackgroundColor3 = theme.Accent,
            Size = UDim2.new(rgb[index] / 255, 0, 1, 0),
            ZIndex = 88,
        }, track)
        radius(fill, 3)

        rowFrame.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                local function applyInput(move)
                    local ratio = math.clamp((move.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
                    rgb[index] = ratio * 255
                    fill.Size = UDim2.new(ratio, 0, 1, 0)
                    applyColor(true)
                end
                applyInput(input)
                DragBus:Begin(applyInput)
            end
        end)
    end

    channel("R", 1, 58)
    channel("G", 2, 88)
    channel("B", 3, 118)

    hex.FocusLost:Connect(function()
        hexToRgb(hex.Text)
    end)

    local catcher = newInstance("TextButton", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false,
        Visible = false,
        ZIndex = 84,
    }, self._window._overlay)

    local open = false
    local function setOpen(value)
        open = value
        if open then
            local vp = getViewport()
            local anchor = preview.AbsolutePosition
            local anchorSize = preview.AbsoluteSize
            local width = math.min(230, vp.X - 20)
            local height = 152
            local x = math.clamp(anchor.X, 10, math.max(vp.X - width - 10, 10))
            local y = anchor.Y + anchorSize.Y + 5
            if y + height > vp.Y - 10 then
                y = anchor.Y - height - 5
            end
            y = math.clamp(y, 10, math.max(vp.Y - height - 10, 10))
            popup.Position = UDim2.fromOffset(x, y)
            popup.Size = UDim2.fromOffset(width, height)
            popup.Visible = true
            catcher.Visible = true
            tween(popup, { BackgroundTransparency = 0 }, 0.12)
        else
            popup.Visible = false
            catcher.Visible = false
        end
    end

    preview.Activated:Connect(function()
        setOpen(not open)
    end)

    catcher.Activated:Connect(function()
        setOpen(false)
    end)

    applyColor(false)

    return {
        Instance = row,
        Get = function()
            return color
        end,
        Set = function(_, value)
            if typeof(value) == "Color3" then
                color = value
                rgb = {
                    math.floor(value.R * 255 + 0.5),
                    math.floor(value.G * 255 + 0.5),
                    math.floor(value.B * 255 + 0.5),
                }
                applyColor(true)
            end
        end,
        Open = function() setOpen(true) end,
        Close = function() setOpen(false) end,
        Destroy = function()
            popup:Destroy()
            catcher:Destroy()
            row:Destroy()
        end,
    }
end

function WindowMeta:CreateTopbarButton(name, icon, callback)
    local button = newInstance("TextButton", {
        BackgroundColor3 = self._theme.Surface,
        Size = UDim2.fromOffset(32, 32),
        Text = "",
        AutoButtonColor = false,
    }, self._header)

    radius(button, 9)
    windowThemeBind(self, button, "BackgroundColor3", "Surface")

    local holder = newInstance("Frame", {
        BackgroundTransparency = 1,
        Size = UDim2.fromOffset(18, 18),
        Position = UDim2.new(0.5, -9, 0.5, -9),
    }, button)
    makeIcon(holder, icon, 15, self._theme.Text)

    bindButtonFeel(button, function() return self._theme.Surface end, function() return self._theme.SurfaceHover end)
    button.Activated:Connect(function()
        safeCall(callback)
    end)

    if not self._topbarButtons then
        self._topbarButtons = {}
    end

    table.insert(self._topbarButtons, button)
    local count = #self._topbarButtons
    local right = 124 + ((count - 1) * 38)
    button.Position = UDim2.new(1, -right, 0.5, -16)

    return button
end

    end
    return self
end

    return self:SetSubtitle(value)
end

function WindowMeta:SetSubtitle(value)
    if self._subtitleLabel then
        self._subtitleLabel.Text = tostring(value)
    end
    return self
end

function WindowMeta:SetIcon(icon)
    local iconFrame = self._brandIcon
    if iconFrame then
        for _, child in ipairs(iconFrame:GetChildren()) do
            if child:IsA("TextLabel") or child:IsA("ImageLabel") then
                child:Destroy()
            end
        end
        makeIcon(iconFrame, icon, 16, self._theme.Accent)
    end
    return self
end

function WindowMeta:SetIconSize(value)
    value = tonumber(value)
    if not value or not self._brandIcon then
        return self
    end
    local icon = self._brandIcon:FindFirstChildWhichIsA("TextLabel") or self._brandIcon:FindFirstChildWhichIsA("ImageLabel")
    if icon then
        icon.Size = UDim2.fromOffset(value, value)
        if icon:IsA("TextLabel") then
            icon.TextSize = math.max(value - 2, 10)
        end
    end
    return self
end

function WindowMeta:SetSize(value)
    if typeof(value) == "UDim2" then
        self._size = value
        tween(self._main, { Size = value }, 0.20)
        clampPosition(self)
    end
    return self
end

function WindowMeta:SetSidebarWidth(value)
    value = tonumber(value)
    if not value then
        return self
    end
    self._sidebarWidth = math.max(value, 140)
    if not self._compact then
        self._sidebar.Size = UDim2.new(0, self._sidebarWidth, 1, 0)
        self._tabHost.Position = UDim2.new(0, self._sidebarWidth, 0, 0)
        self._tabHost.Size = UDim2.new(1, -self._sidebarWidth, 1, 0)
    end
    return self
end

function WindowMeta:SetToggleKey(key)
    if self._toggleUnregister then
        self._toggleUnregister()
        self._toggleUnregister = nil
    end
    if key then
        self._toggleUnregister = InputBus:RegisterToggle(key, function()
            self:Toggle()
        end)
    end
    return self
end

function WindowMeta:Toggle()
    if self._destroyed then
        return self
    end
    if self._visible then
        self:Close()
    else
        self:Open()
    end
    return self
end

function WindowMeta:Open()
    if self._destroyed or self._visible then
        return self
    end
    self._visible = true
    self._main.Visible = true
    self._scale.Scale = 0.96
    tween(self._scale, { Scale = 1 }, 0.20)
    for _, callback in ipairs(self._onOpen) do
        safeCall(callback)
    end
    return self
end

function WindowMeta:Close()
    if self._destroyed or not self._visible then
        return self
    end
    self._visible = false
    local t = tween(self._scale, { Scale = 0.96 }, 0.16)
    if t then
        t.Completed:Connect(function()
            if not self._destroyed and not self._visible then
                self._main.Visible = false
            end
        end)
    else
        self._main.Visible = false
    end
    for _, callback in ipairs(self._onClose) do
        safeCall(callback)
    end
    return self
end

function WindowMeta:Minimize()
    if self._destroyed or not self._minimizable then
        return self
    end

    self._minimized = not self._minimized
    if self._minimized then
        self._preMinimizeSize = self._main.Size
        tween(self._main, { Size = UDim2.fromOffset(self._main.AbsoluteSize.X, self._metrics.TopBar) }, 0.20)
    else
        tween(self._main, { Size = self._preMinimizeSize or self._size }, 0.20)
    end
    return self
end

function WindowMeta:Maximize()
    if self._destroyed then
        return self
    end
    self._minimized = false
    self._fullscreen = true
    local vp = getViewport()
    tween(self._main, {
        Size = UDim2.fromOffset(math.max(vp.X - 20, 300), math.max(vp.Y - 20, 260)),
        Position = UDim2.fromOffset(10, 10),
    }, 0.28)
    return self
end

function WindowMeta:Restore()
    if self._destroyed then
        return self
    end
    self._fullscreen = false
    self._minimized = false
    tween(self._main, {
        Size = self._size,
        Position = centerPosition(self._size),
    }, 0.25)
    return self
end

function WindowMeta:ToggleFullscreen()
    if self._fullscreen then
        return self:Restore()
    end
    return self:Maximize()
end

function WindowMeta:SetTheme(theme)
    if type(theme) == "string" then
        theme = VeloxUI.Themes[theme]
    end
    if type(theme) ~= "table" then
        return self
    end

    self._theme = mergeTheme(DEFAULTS, theme)

    for _, bind in ipairs(self._themeBinds) do
        if bind.instance and bind.instance.Parent and self._theme[bind.key] ~= nil then
            pcall(function()
                bind.instance[bind.property] = self._theme[bind.key]
            end)
        end
    end

    for _, tab in ipairs(self._tabs) do
        tab._theme = self._theme
    end

    return self
end

function WindowMeta:Dialog(config)
    config = config or {}
    local theme = self._theme

    local overlay = newInstance("Frame", {
        BackgroundColor3 = Color3.new(0, 0, 0),
        BackgroundTransparency = 0.45,
        Size = UDim2.fromScale(1, 1),
        ZIndex = 200,
    }, self._gui)

    local card = newInstance("Frame", {
        BackgroundColor3 = theme.Surface,
        Size = UDim2.fromOffset(math.min(config.Width or 360, getViewport().X - 32), 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0.5, 0.5),
        ZIndex = 201,
    }, overlay)
    radius(card, 14)
    outline(card, theme.Border, 1)
    addPadding(card, 18, 16, 18, 16)

    local title = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, 24),
        Font = Enum.Font.GothamBold,
        Text = tostring(config.Title or "Dialog"),
        TextColor3 = theme.Text,
        TextSize = 17,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 202,
    }, card)

    local content = newInstance("TextLabel", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 31),
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        Font = Enum.Font.Gotham,
        Text = tostring(config.Content or ""),
        TextColor3 = theme.SubText,
        TextSize = 13,
        TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 202,
    }, card)

    local buttons = newInstance("Frame", {
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 78),
        Size = UDim2.new(1, 0, 0, 38),
        ZIndex = 202,
    }, card)

    newInstance("UIListLayout", {
        FillDirection = Enum.FillDirection.Horizontal,
        HorizontalAlignment = Enum.HorizontalAlignment.Right,
        Padding = UDim.new(0, 8),
    }, buttons)

    local closeDialog
    local function makeButton(def)
        local btn = newInstance("TextButton", {
            BackgroundColor3 = def.Variant == "Danger" and theme.Negative or theme.Accent,
            Size = UDim2.fromOffset(100, 36),
            Text = tostring(def.Title or "OK"),
            Font = Enum.Font.GothamBold,
            TextColor3 = Color3.new(1, 1, 1),
            TextSize = 12,
            AutoButtonColor = false,
            ZIndex = 203,
        }, buttons)
        radius(btn, 9)

        if def.Variant == "Secondary" then
            btn.BackgroundColor3 = theme.SurfaceHover
            btn.TextColor3 = theme.Text
            outline(btn, theme.Border, 1)
        end

        btn.Activated:Connect(function()
            safeCall(def.Callback)
            if def.Close ~= false then
                closeDialog()
            end
        end)

        return btn
    end

    closeDialog = function()
        if overlay and overlay.Parent then
            overlay:Destroy()
        end
    end

    for _, def in ipairs(config.Buttons or {
        { Title = "Close", Variant = "Secondary" },
    }) do
        makeButton(def)
    end

    return {
        Instance = overlay,
        Close = closeDialog,
        Destroy = closeDialog,
    }
end

function WindowMeta:Destroy()
    if self._destroyed then
        return
    end

    self._destroyed = true
    DragBus:Cancel()

    if self._toggleUnregister then
        self._toggleUnregister()
        self._toggleUnregister = nil
    end

    for _, callback in ipairs(self._onDestroy) do
        safeCall(callback)
    end

    self._cleaner:Clean()

    if self._gui then
        self._gui:Destroy()
    end
end

local function addControlAliases(tab)
    tab.Button = function(self, config) return self:CreateButton(config) end
    tab.Toggle = function(self, config) return self:CreateToggle(config) end
    tab.Slider = function(self, config) return self:CreateSlider(config) end
    tab.Dropdown = function(self, config) return self:CreateDropdown(config) end
    tab.Input = function(self, config) return self:CreateInput(config) end
    tab.Keybind = function(self, config) return self:CreateKeybind(config) end
    tab.Colorpicker = function(self, config) return self:CreateColorpicker(config) end
    tab.Paragraph = function(self, config) return self:CreateParagraph(config) end
    tab.Divider = function(self) return self:CreateDivider() end
end

function VeloxUI:SetTheme(name, theme)
    if type(name) == "table" and theme == nil then
        theme = name
        name = "Custom"
    end

    if type(name) == "string" and type(theme) == "table" then
        self.Themes[name] = mergeTheme(DEFAULTS, theme)
    end

    if type(name) == "string" and self.Themes[name] then
        self.CurrentTheme = self.Themes[name]
    elseif type(theme) == "table" then
        self.CurrentTheme = mergeTheme(DEFAULTS, theme)
    end

    return self.CurrentTheme
end

function VeloxUI:AddTheme(name, theme)
    if type(name) == "string" and type(theme) == "table" then
        self.Themes[name] = mergeTheme(DEFAULTS, theme)
        return self.Themes[name]
    end
end

function VeloxUI:CreateTheme(name, theme)
    return self:AddTheme(name, theme)
end

function VeloxUI:CreateWindow(config)
    local window = self:CreateWindowBase(config)
    local originalCreateTab = window.CreateTab

    function window:CreateTab(tabConfig)
        local tab = originalCreateTab(self, tabConfig)
        addControlAliases(tab)
        return tab
    end

    window.Tab = window.CreateTab

    window._setDrawer = function(open)
        if window._compact and window._backdrop and window._sidebar then
            window._backdrop.Visible = open
            local targetX = open and 0 or -window._sidebar.AbsoluteSize.X
            tween(window._backdrop, { BackgroundTransparency = open and 0.4 or 1 }, 0.18)
            tween(window._sidebar, { Position = UDim2.new(0, targetX, 0, 0) }, 0.20)
        end
    end

    return window
end

VeloxUI.Window = VeloxUI.CreateWindow

function VeloxUI:DestroyAll()
    local core = game:GetService("CoreGui")
    for _, child in ipairs(core:GetChildren()) do
        if child:IsA("ScreenGui") and (child.Name == "VeloxUI" or child.Name == "VeloxUI_Notifications") then
            pcall(function()
                child:Destroy()
            end)
        end
    end

    local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
    if playerGui then
        for _, child in ipairs(playerGui:GetChildren()) do
            if child:IsA("ScreenGui") and (child.Name == "VeloxUI" or child.Name == "VeloxUI_Notifications") then
                pcall(function()
                    child:Destroy()
                end)
            end
        end
    end
end
